126 [[nodiscard]] constexpr containers::inplace_vector<std::uint8_t, 72> der() const noexcept
127 {
128 // Each integer appended straight from r or s. Appending them through intermediate vectors made GCC 15
129 // report false overflows (-Wstringop-overflow at -O3 with AVX2), and writing through a running index into a
130 // result sized to its capacity did the same under the sanitizers (-Warray-bounds): GCC cannot relate the two
131 // lengths to their sum of at most 72. Nothing here can find the result full: 2 + 35 + 35 bytes at most.
132 containers::inplace_vector<std::uint8_t, 72> out;
133 auto integer = [&out](const std::array<std::uint8_t, 32>& v) {
134 std::size_t i = 0;
135 while (i < 31 && v[i] == 0)
136 ++i;
137 const std::size_t sign = v[i] >> 7; // a zero byte before a first byte with its top bit set
138 (void)out.try_push_back(0x02);
139 (void)out.try_push_back(static_cast<std::uint8_t>(sign + 32 - i));
140 if (sign != 0)
141 (void)out.try_push_back(0);
142 for (; i < 32; ++i)
143 (void)out.try_push_back(v[i]);
144 };
145 (void)out.try_push_back(0x30);
146 (void)out.try_push_back(0); // the SEQUENCE's length, known at the end
147 integer(r);
148 integer(s);
149 out[1] = static_cast<std::uint8_t>(out.size() - 2);
150 return out;
151 }

156 [[nodiscard]] static constexpr result<signature, errc> from_der(std::span<const std::uint8_t> der) noexcept
157 {
158 const std::size_t n = der.size();
159 if (n < 8 || n > 72 || der[0] != 0x30 || der[1] != n - 2 || der[2] != 0x02)
160 return err(errc::invalid_signature);
161 const std::size_t len_r = der[3];
162 if (len_r == 0 || 5 + len_r >= n || der[4 + len_r] != 0x02)
163 return err(errc::invalid_signature);
164 const std::size_t len_s = der[5 + len_r];
165 if (len_s == 0 || len_r + len_s + 6 != n)
166 return err(errc::invalid_signature);
167 signature sig;
168 auto integer = [](std::span<const std::uint8_t> v, std::array<std::uint8_t, 32>& out) {
169 if (v[0] & 0x80)
170 return false; // negative
171 if (v.size() > 1 && v[0] == 0 && !(v[1] & 0x80))
172 return false; // padding beyond the sign byte
173 if (v.size() > 33 || (v.size() == 33 && v[0] != 0))
174 return false; // more than 256 bits
175 const std::size_t skip = v.size() == 33 ? 1 : 0;
176 const std::size_t pad = 32 - (v.size() - skip);
177 for (std::size_t i = 0; i < v.size() - skip; ++i)
178 out[pad + i] = v[skip + i];
179 return true;
180 };
181 if (!integer(der.subspan(4, len_r), sig.r) || !integer(der.subspan(6 + len_r, len_s), sig.s))
182 return err(errc::invalid_signature);
183 return ok(sig);
184 }

225[[nodiscard]] constexpr recoverable_signature sign(const private_key& key, std::span<const std::uint8_t, 32> hash,
226 std::span<const std::uint8_t> extra = {}) noexcept
227{
228 const auto order = scalar::order_bytes();
229 scalar d = key.to_scalar();
230 scalar z = scalar::from_bytes_reduced(hash);
231 rfc6979 nonces(key.reveal(), hash, order, extra);
232 for (;;) {
233 const secret_bytes<32> kb = nonces.next();
234 scalar k = scalar::from_bytes_reduced(kb.reveal());
235 std::uint64_t finite = 0;
236 const affine_point big_r = point::mul_base(k).to_affine_ct(finite);
237 std::uint64_t below = 0;
238 const scalar r = scalar::from_bytes_ct(big_r.x.to_bytes(), below); // r = x mod n; overflow when x >= n
239 scalar k_inverse = k.inverse();
240 scalar s = k_inverse * (z + r * d);
241 const std::uint64_t high = s.is_high_mask();
242 s = scalar::select(high, -s, s);
243 // The recovery id is published with the signature: its two bits are public from here on
244 const std::uint64_t parity = ct::declassify(big_r.y.is_odd() ^ (high & 1));
245 const std::uint64_t overflow = ct::declassify(~below & 1);
246 const auto id = static_cast<std::uint8_t>(parity | (overflow << 1));
247 secure_wipe(k);
248 secure_wipe(k_inverse);
249 // r or s of zero, with probability about 2^-256: round again with the next RFC 6979 nonce. The loop has no
250 // `continue`: MSVC takes one in an endless loop for control reaching the end of a constexpr function (C3615).
251 if (!ct::declassify(r.is_zero_mask() | s.is_zero_mask())) {
252 secure_wipe(d);
253 secure_wipe(z);
254 return {{r.to_bytes(), s.to_bytes()}, id};
255 }
256 }

260// r and s in [1, n - 1], and s's policy; public data
261constexpr result<std::array<scalar, 2>, errc> signature_scalars(const signature& sig, high_s policy) noexcept
262{
263 const auto r = scalar::from_bytes(sig.r);
264 const auto s = scalar::from_bytes(sig.s);
265 if (!r || !s || r->is_zero() || s->is_zero())
266 return err(errc::invalid_signature);
267 if (policy == high_s::reject && s->is_high())
268 return err(errc::high_s);
269 return ok(std::array<scalar, 2>{*r, *s});
270}

280[[nodiscard]] constexpr result<void, errc> verify(const public_key& key, std::span<const std::uint8_t, 32> hash,
281 const signature& sig, high_s policy = high_s::accept) noexcept
282{
283 const auto rs = detail::signature_scalars(sig, policy);
284 if (!rs)
285 return err(rs.unwrap_err());
286 const scalar r = (*rs)[0], s = (*rs)[1];
287 const scalar z = scalar::from_bytes_reduced(hash);
288 const scalar w = s.inverse();
289 const auto x = point::mul_add_public(z * w, r * w, point(key.point())).to_affine();
290 if (!x || !(scalar::from_bytes_reduced(x->x.to_bytes()) == r))
291 return err(errc::invalid_signature);
292 return ok();
293}

302[[nodiscard]] constexpr result<public_key, errc> recover(std::span<const std::uint8_t, 32> hash, const signature& sig,
303 std::uint8_t recovery_id, high_s policy = high_s::accept) noexcept
304{
305 if (recovery_id > 3)
306 return err(errc::invalid_recovery_id);
307 const auto rs = detail::signature_scalars(sig, policy);
308 if (!rs)
309 return err(rs.unwrap_err());
310 const scalar r = (*rs)[0], s = (*rs)[1];
311 // x = r + (id / 2) n, which must be a field element
312 auto x_limbs = crypto::detail::from_be(sig.r);
313 if (recovery_id & 2) {
314 crypto::detail::limbs4 t{};
315 if (crypto::detail::add4(t, x_limbs, secp256k1::detail::order_modulus.m))
316 return err(errc::recovery_failed);
317 x_limbs = t;
318 }
319 const auto x = field_element::from_bytes(crypto::detail::to_be(x_limbs));
320 if (!x)
321 return err(errc::recovery_failed);
322 const auto big_r = decompress(*x, recovery_id & 1);
323 if (!big_r)
324 return err(errc::recovery_failed);
325 const scalar z = scalar::from_bytes_reduced(hash);
326 const scalar r_inverse = r.inverse();
327 const auto q = point::mul_add_public(-(z * r_inverse), s * r_inverse, point(*big_r)).to_affine();
328 if (!q)
329 return err(errc::recovery_failed);
330 return ok(public_key(*q));
331}

340[[nodiscard]] constexpr result<public_key, errc> recover_ethereum(std::span<const std::uint8_t, 32> hash,
341 std::span<const std::uint8_t, 65> rsv,
342 high_s policy = high_s::accept) noexcept
343{
344 if (rsv[64] != 27 && rsv[64] != 28)
345 return err(errc::invalid_recovery_id);
346 return recover(hash, signature::from_compact(rsv.first<64>()), static_cast<std::uint8_t>(rsv[64] - 27), policy);
347}
