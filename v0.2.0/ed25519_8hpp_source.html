147 [[nodiscard]] constexpr signature sign(std::span<const std::uint8_t> message) const noexcept
148 {
149 crypto::detail::sha512 h;
150 h.update(prefix_.reveal());
151 h.update(message);
152 auto rh = h.finish();
153 curve25519::scalar r = curve25519::scalar::reduce(rh);
154 secure_wipe(rh);
155 const auto big_r = curve25519::point::mul_base(r).encode();
156
157 crypto::detail::sha512 hk;
158 hk.update(big_r);
159 hk.update(public_.bytes);
160 hk.update(message);
161 const auto kh = hk.finish();
162 const curve25519::scalar k = curve25519::scalar::reduce(kh);
163
164 const curve25519::scalar a = curve25519::scalar::from_bytes_reduced(scalar_.reveal());
165 curve25519::scalar s = curve25519::scalar::mul_add(k, a, r);
166 signature sig;
167 for (std::size_t i = 0; i < 32; ++i) {
168 sig.bytes[i] = big_r[i];
169 sig.bytes[32 + i] = s.to_bytes()[i];
170 }
171 secure_wipe(r);
172 secure_wipe(s);
173 auto a_copy = a;
174 secure_wipe(a_copy);
175 return sig;
176 }

178private:
179 constexpr explicit key_pair(secret_bytes<32>&& seed) noexcept : seed_(std::move(seed))
180 {
181 auto d = crypto::detail::sha512::hash(seed_.reveal());
182 std::array<std::uint8_t, 32> a{};
183 std::array<std::uint8_t, 32> prefix{};
184 for (std::size_t i = 0; i < 32; ++i) {
185 a[i] = d[i];
186 prefix[i] = d[32 + i];
187 }
188 secure_wipe(d);
189 a[0] = static_cast<std::uint8_t>(a[0] & 248);
190 a[31] = static_cast<std::uint8_t>((a[31] & 127) | 64);
191 // The clamped value is below 2^255; B has order L, so the scalar can be held reduced modulo L
192 curve25519::scalar s = curve25519::scalar::from_bytes_reduced(a);
193 secure_wipe(a);
194 public_.bytes = curve25519::point::mul_base(s).encode();
195 auto sb = s.to_bytes();
196 scalar_ = secret_bytes<32>(std::move(sb));
197 prefix_ = secret_bytes<32>(std::move(prefix));
198 secure_wipe(s);
199 }

220[[nodiscard]] constexpr snicholls::result<void, errc> verify(const public_key& key, std::span<const std::uint8_t> message,
221 const signature& sig) noexcept
222{
223 const auto a = curve25519::point::decode(key.bytes);
224 if (!a)
225 return snicholls::err(errc::invalid_public_key);
226 const auto sig_span = std::span<const std::uint8_t, 64>(sig.bytes);
227 const auto s = curve25519::scalar::from_canonical(sig_span.subspan<32, 32>());
228 if (!s)
229 return snicholls::err(errc::invalid_signature);
230 crypto::detail::sha512 hk;
231 hk.update(sig_span.first<32>());
232 hk.update(key.bytes);
233 hk.update(message);
234 const curve25519::scalar k = curve25519::scalar::reduce(hk.finish());
235 const auto check = curve25519::point::mul_add_public(*s, k, -*a).encode();
236 for (std::size_t i = 0; i < 32; ++i)
237 if (check[i] != sig.bytes[i])
238 return snicholls::err(errc::invalid_signature);
239 return snicholls::ok();
240}
