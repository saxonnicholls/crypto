108 [[nodiscard]] constexpr result<slip10_ed25519_key, errc> derive_child(std::uint32_t index) const noexcept
109 {
110 if (!is_hardened(index))
111 return err(errc::hardened_required);
112 crypto::detail::hmac_sha512 m(chain_.reveal());
113 const std::uint8_t zero = 0;
114 m.update(std::span<const std::uint8_t>(&zero, 1));
115 m.update(key_.reveal());
116 m.update(detail::be32(index));
117 return ok(from_hmac(m.finish(), static_cast<std::uint8_t>(depth_ + 1), fingerprint(), index));
118 }

123 [[nodiscard]] constexpr result<slip10_ed25519_key, errc> derive(const derivation_path& path) const noexcept
124 {
125 if (!path.all_hardened())
126 return err(errc::hardened_required);
127 slip10_ed25519_key k = clone();
128 for (const std::uint32_t index : path.indices())
129 k = k.derive_child(index).unwrap_unchecked();
130 return ok(std::move(k));
131 }

160 {
161 std::array<std::uint8_t, 33> pk{};
162 const auto p = public_key();
163 for (std::size_t i = 0; i < 32; ++i)
164 pk[1 + i] = p.bytes[i];
