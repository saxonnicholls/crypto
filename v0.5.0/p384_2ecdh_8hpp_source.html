80[[nodiscard]] constexpr result<secret_bytes<48>, errc> shared_secret(const private_key& key, const public_key& peer) noexcept
81{
82 if (!peer.point().is_on_curve()) // public data
83 return err(errc::invalid_public_key);
84 scalar d = key.to_scalar();
85 std::uint64_t finite = 0;
86 affine_point s = point::mul(d, point(peer.point())).to_affine_ct(finite);
87 secure_wipe(d);
88 std::array<std::uint8_t, 48> x = s.x.to_bytes();
89 secure_wipe(s);
90 if (!ct::declassify(finite != 0)) {
91 secure_wipe(x);
92 return err(errc::invalid_public_key);
93 }
94 return ok(secret_bytes<48>(std::move(x)));
95}
