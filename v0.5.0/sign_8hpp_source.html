122// The private key's CRT form at N limbs a prime: the primes with their Montgomery constants, dP, dQ, and qInv
123// (q^-1 mod p). All of it is secret
124template <std::size_t N>
125struct crt_key {
126 mont_params<N> p{};
127 mont_params<N> q{};
128 limbs_n<N> dp{};
129 limbs_n<N> dq{};
130 limbs_n<N> qinv{};
131};

133template <std::size_t N>
134using limbs_2n = std::array<std::uint64_t, 2 * N>;
135
136// x R mod m (x in Montgomery form) for any x below 2^(128 N), x = hi 2^(64 N) + lo. A Montgomery multiplication by
137// R^2 mod m is fully reduced for any other operand below R, so lo R and (hi R) R come out below m
138template <std::size_t N>
139constexpr limbs_n<N> residue_mont(const limbs_2n<N>& x, const mont_params<N>& m) noexcept
140{
141 limbs_n<N> lo{}, hi{};
142 for (std::size_t i = 0; i < N; ++i) {
143 lo[i] = x[i];
144 hi[i] = x[N + i];
145 }
146 limbs_n<N> a = crypto::detail::mont_mul<N>(lo, m.r2, m);
147 limbs_n<N> b = crypto::detail::mont_mul<N>(crypto::detail::mont_mul<N>(hi, m.r2, m), m.r2, m);
148 const limbs_n<N> r = crypto::detail::mont_add<N>(a, b, m);
149 secure_wipe(lo);
150 secure_wipe(hi);
151 secure_wipe(a);
152 secure_wipe(b);
153 return r;
154}

156// a b, all 2 N limbs, in constant time
157template <std::size_t N>
158constexpr limbs_2n<N> mul_wide(const limbs_n<N>& a, const limbs_n<N>& b) noexcept
159{
160 limbs_2n<N> r{};
161 for (std::size_t i = 0; i < N; ++i) {
162 std::uint64_t carry = 0;
163 for (std::size_t j = 0; j < N; ++j) {
164 const auto prod = snicholls::detail::umul128(a[i], b[j]);
165 std::uint64_t lo = prod.low, hi = prod.high, c = 0;
166 lo = crypto::detail::add_carry(lo, r[i + j], c);
167 hi += c;
168 c = 0;
169 lo = crypto::detail::add_carry(lo, carry, c);
170 hi += c;
171 r[i + j] = lo;
172 carry = hi;
173 }
174 r[i + N] = carry;
175 }
176 return r;
177}

179// m^d mod n by the CRT (RFC 8017 5.1.2, step 2.b), for m < n = p q:
180// m1 = m^dP mod p, m2 = m^dQ mod q, h = qInv (m1 - m2) mod p, s = m2 + q h
181template <std::size_t N>
182constexpr limbs_2n<N> crt_power(const crt_key<N>& k, const limbs_2n<N>& m) noexcept
183{
184 using namespace crypto::detail;
185 limbs_n<N> mp = residue_mont<N>(m, k.p);
186 limbs_n<N> mq = residue_mont<N>(m, k.q);
187 limbs_n<N> m1 = mont_pow_ct<N>(mp, k.dp, k.p); // m1 R mod p
188 limbs_n<N> m2r = mont_pow_ct<N>(mq, k.dq, k.q); // m2 R mod q
189 limbs_n<N> m2 = mont_from<N>(m2r, k.q); // m2, below q
190 limbs_n<N> m2p = mont_mul<N>(m2, k.p.r2, k.p); // m2 R mod p (m2 may be above p: it is below R)
191 limbs_n<N> diff = mont_sub<N>(m1, m2p, k.p); // (m1 - m2) R mod p
192 limbs_n<N> h = mont_mul<N>(diff, k.qinv, k.p); // (m1 - m2) qInv mod p, out of Montgomery form
193 limbs_2n<N> s = mul_wide<N>(k.q.m, h); // q h < q p = n
194 std::uint64_t carry = 0;
195 for (std::size_t i = 0; i < 2 * N; ++i) // + m2 < q: s < n, no carry out
196 s[i] = add_carry(s[i], i < N ? m2[i] : 0, carry);
197 secure_wipe(mp);
198 secure_wipe(mq);
199 secure_wipe(m1);
200 secure_wipe(m2r);
201 secure_wipe(m2);
202 secure_wipe(m2p);
203 secure_wipe(diff);
204 secure_wipe(h);
205 return s;
206}

208// Big-endian bytes (at most 8 L of them) into L limbs, zero above; the bytes may be secret, their number is public
209template <std::size_t L>
210constexpr std::array<std::uint64_t, L> limbs_from_be(std::span<const std::uint8_t> b) noexcept
211{
212 std::array<std::uint64_t, L> r{};
213 for (std::size_t i = 0; i < b.size(); ++i) {
214 const std::size_t bit = 8 * (b.size() - 1 - i);
215 r[bit / 64] |= static_cast<std::uint64_t>(b[i]) << (bit % 64);
216 }
217 return r;
218}

220// a < b as a mask
221template <std::size_t N>
222constexpr std::uint64_t mask_less_n(const limbs_n<N>& a, const limbs_n<N>& b) noexcept
223{
224 limbs_n<N> t{};
225 const std::uint64_t borrow = crypto::detail::sub_n<N>(t, a, b);
226 secure_wipe(t);
227 return ct::mask_from_bit(borrow);
228}

230// A component read where its bytes are secret: non-empty and without a leading zero byte. The test is on the first
231// byte only and decides the component's length, which is public
232constexpr bool canonical_secret(std::span<const std::uint8_t> v) noexcept
233{
234 return !v.empty() && ct::declassify(v[0] != 0);
235}
236
237// A DER INTEGER's contents, secret, as its unsigned magnitude: non-negative and minimal (der_unsigned, with the
238// tests on its first two bytes declared: they are DER's form, not the number)
239constexpr std::optional<std::span<const std::uint8_t>> der_unsigned_secret(std::span<const std::uint8_t> v) noexcept
240{
241 if (v.empty() || ct::declassify((v[0] & 0x80) != 0))
242 return std::nullopt;
243 if (v.size() == 1)
244 return v;
245 const bool zero = ct::declassify(v[0] == 0);
246 if (zero && !ct::declassify((v[1] & 0x80) != 0))
247 return std::nullopt;
248 return zero ? v.subspan(1) : v;
249}

285 [[nodiscard]] static constexpr result<private_key, errc> from_pkcs1_der(std::span<const std::uint8_t> der) noexcept
286 {
287 detail::der_reader outer(der);
288 const auto seq = outer.next(0x30);
289 if (!seq || !outer.done())
290 return err(errc::invalid_private_key);
291 detail::der_reader in(*seq);
292 const auto version = in.next(0x02);
293 if (!version || version->size() != 1 || ct::declassify((*version)[0] != 0))
294 return err(errc::invalid_private_key); // 0: two primes; 1 (multi-prime) is not read
295 std::array<std::span<const std::uint8_t>, 8> v{}; // n, e, d, p, q, dP, dQ, qInv
296 for (auto& x : v) {
297 const auto raw = in.next(0x02);
298 const auto u = raw ? detail::der_unsigned_secret(*raw) : std::nullopt;
299 if (!u)
300 return err(errc::invalid_private_key);
301 x = *u;
302 }
303 if (!in.done())
304 return err(errc::invalid_private_key);
305 // d (v[2]) is not used: the CRT exponents are, and the consistency test proves them against e
306 auto pub = public_key::from_components(v[0], v[1]);
307 if (!pub)
308 return err(errc::invalid_private_key);
309 for (std::size_t i = 3; i < 8; ++i)
310 if (!detail::canonical_secret(v[i]))
311 return err(errc::invalid_private_key);
312 return make(*pub, v[0], v[3], v[4], v[5], v[6], v[7]);
313 }

320 [[nodiscard]] static constexpr result<private_key, errc> from_pkcs8_der(std::span<const std::uint8_t> der) noexcept
321 {
322 detail::der_reader outer(der);
323 const auto seq = outer.next(0x30);
324 if (!seq || !outer.done())
325 return err(errc::invalid_private_key);
326 detail::der_reader in(*seq);
327 const auto version = in.next(0x02);
328 const auto alg = version ? in.next(0x30) : std::nullopt;
329 const auto key = alg ? in.next(0x04) : std::nullopt;
330 if (!key || !in.done() || version->size() != 1 || (*version)[0] != 0)
331 return err(errc::invalid_private_key);
332 detail::der_reader a(*alg);
333 const auto oid = a.next(0x06);
334 const auto null = oid ? a.next(0x05) : std::nullopt;
335 if (!null || !null->empty() || !a.done() || oid->size() != detail::rsa_encryption_oid.size())
336 return err(errc::invalid_private_key);
337 for (std::size_t i = 0; i < oid->size(); ++i)
338 if ((*oid)[i] != detail::rsa_encryption_oid[i])
339 return err(errc::invalid_private_key);
340 return from_pkcs1_der(*key);
341 }

432 // The checks of a key at N limbs a prime, as one mask: p and q odd and different, p q = n, 0 < dP < p,
433 // 0 < dQ < q, 0 < qInv < p, qInv q = 1 mod p
434 template <std::size_t N>
435 constexpr std::uint64_t consistent(std::span<const std::uint8_t> n) const noexcept
436 {
437 using namespace crypto::detail;
438 limbs_n<N> p = low<N>(p_), q = low<N>(q_), dp = low<N>(dp_), dq = low<N>(dq_), qinv = low<N>(qinv_);
439 std::uint64_t good = ct::mask_from_bit(p[0] & q[0] & 1);
440 good &= ~mask_equal_n<N>(p, q);
441 auto pq = detail::mul_wide<N>(p, q);
442 const auto nl = detail::limbs_from_be<2 * N>(n);
443 good &= mask_equal_n<2 * N>(pq, nl);
444 good &= detail::mask_less_n<N>(dp, p) & ~mask_is_zero_n<N>(dp);
445 good &= detail::mask_less_n<N>(dq, q) & ~mask_is_zero_n<N>(dq);
446 good &= detail::mask_less_n<N>(qinv, p) & ~mask_is_zero_n<N>(qinv);
447 // with an even p there is no Montgomery form; the product below is then meaningless and good is already zero
448 mont_params<N> mp = make_mont_ct<N>(p);
449 limbs_n<N> qr = mont_mul<N>(q, mp.r2, mp); // q R mod p
450 limbs_n<N> one = mont_mul<N>(qr, qinv, mp); // q qInv mod p
451 limbs_n<N> expect{};
452 expect[0] = 1;
453 good &= mask_equal_n<N>(one, expect);
454 secure_wipe(p);
455 secure_wipe(q);
456 secure_wipe(dp);
457 secure_wipe(dq);
458 secure_wipe(qinv);
459 secure_wipe(pq);
460 secure_wipe(mp);
461 secure_wipe(qr);
462 secure_wipe(one);
463 return good;
464 }

475 template <std::size_t N>
476 constexpr result<void, errc> sign_n(std::span<const std::uint8_t> em, std::span<std::uint8_t> out) const noexcept
477 {
478 using namespace crypto::detail;
479 const std::size_t k = modulus_bytes();
480 detail::crt_key<N> key;
481 key.p = make_mont_ct<N>(low<N>(p_));
482 key.q = make_mont_ct<N>(low<N>(q_));
483 key.dp = low<N>(dp_);
484 key.dq = low<N>(dq_);
485 key.qinv = low<N>(qinv_);
486 auto s = detail::crt_power<N>(key, detail::limbs_from_be<2 * N>(em));
487 secure_wipe(key);
488 std::array<std::uint8_t, max_modulus_bytes> sig{};
489 for (std::size_t i = 0; i < k; ++i) {
490 const std::size_t bit = 8 * (k - 1 - i);
491 sig[i] = static_cast<std::uint8_t>(s[bit / 64] >> (bit % 64));
492 }
493 secure_wipe(s);
494 // The signature is public from here: it is what this function returns. Declared so, it is verified with the
495 // public key (variable time); one that does not verify is wiped and never leaves
496 sig = ct::declassify(sig);
497 std::array<std::uint8_t, max_modulus_bytes> check{};
498 bool same = pub_.open(std::span<const std::uint8_t>(sig).first(k), check).is_ok();
499 std::uint8_t diff = 0;
500 for (std::size_t i = 0; i < k; ++i)
501 diff = static_cast<std::uint8_t>(diff | (check[i] ^ em[i]));
502 same = same && diff == 0;
503 for (std::size_t i = 0; i < k; ++i)
504 out[i] = same ? sig[i] : std::uint8_t{0};
505 secure_wipe(sig);
506 if (!same)
507 return err(errc::signing_fault);
508 return ok();
509 }

511 static constexpr result<private_key, errc> make(const rsa::public_key& pub, std::span<const std::uint8_t> n,
512 std::span<const std::uint8_t> p, std::span<const std::uint8_t> q,
513 std::span<const std::uint8_t> dp, std::span<const std::uint8_t> dq,
514 std::span<const std::uint8_t> qinv) noexcept
515 {
516 // The primes' byte lengths are public and choose the limb count
517 const std::size_t widest = p.size() > q.size() ? p.size() : q.size();
518 const std::size_t limbs = widest <= 128 ? 16 : widest <= 192 ? 24 : 32;
519 if (widest > 8 * detail::crt_max_limbs || dp.size() > 8 * limbs || dq.size() > 8 * limbs || qinv.size() > 8 * limbs ||
520 n.size() > 16 * limbs)
521 return err(errc::invalid_private_key);
522 private_key key(pub, limbs, detail::limbs_from_be<detail::crt_max_limbs>(p), detail::limbs_from_be<detail::crt_max_limbs>(q),
523 detail::limbs_from_be<detail::crt_max_limbs>(dp), detail::limbs_from_be<detail::crt_max_limbs>(dq),
524 detail::limbs_from_be<detail::crt_max_limbs>(qinv));
525 if (!ct::declassify(key.consistent(n) != 0))
526 return err(errc::invalid_private_key);
527 // The pairwise consistency test: a signature over a fixed encoded message must verify. It proves dP and dQ
528 // against e (the checks above do not), with the same code that signs
529 std::array<std::uint8_t, max_modulus_bytes> em{}, sig{};
530 const std::size_t k = pub.modulus_bytes();
531 for (std::size_t i = 1; i < k; ++i)
532 em[i] = 0x5a; // the first byte is zero: below the modulus
533 const bool signs = key.sign_encoded(std::span<const std::uint8_t>(em).first(k), std::span<std::uint8_t>(sig).first(k)).is_ok();
534 secure_wipe(sig);
535 if (!signs)
536 return err(errc::invalid_private_key);
537 return ok(std::move(key));
538 }

551// EMSA-PKCS1-v1_5 (RFC 8017 9.2) over a hash, then RSASP1
552template <class Scheme>
553constexpr result<void, errc> sign_pkcs1v15(const private_key& key, std::span<const std::uint8_t, Scheme::hash::digest_size> hash,
554 std::span<std::uint8_t> sig) noexcept
555{
556 const std::size_t k = key.modulus_bytes();
557 if (sig.size() != k)
558 return err(errc::invalid_length);
559 // EM = 0x00 0x01 0xFF ... 0xFF 0x00 DigestInfo H (k >= 256, so far more than 8 bytes of 0xFF)
560 std::array<std::uint8_t, max_modulus_bytes> em{};
561 const std::size_t t = Scheme::digest_info.size() + hash.size();
562 std::size_t i = 0;
563 em[i++] = 0x00;
564 em[i++] = 0x01;
565 for (; i < k - t - 1; ++i)
566 em[i] = 0xff;
567 em[i++] = 0x00;
568 for (const auto b : Scheme::digest_info)
569 em[i++] = b;
570 for (const auto b : hash)
571 em[i++] = b;
572 return key.sign_encoded(std::span<const std::uint8_t>(em).first(k), sig);
573}
