100constexpr void pad16(crypto::detail::poly1305& mac, std::size_t n) noexcept
101{
102 const std::array<std::uint8_t, 16> zeros{};
103 if (n % 16 != 0)
104 mac.update(std::span<const std::uint8_t>(zeros).first(16 - n % 16));
105}

107constexpr tag compute_tag(std::span<const std::uint8_t, 32> key, std::span<const std::uint8_t, 12> nonce,
108 std::span<const std::uint8_t> aad, std::span<const std::uint8_t> ciphertext) noexcept
109{
110 auto block = crypto::detail::chacha20_block(key, 0, nonce);
111 crypto::detail::poly1305 mac(std::span<const std::uint8_t, 32>(block.data(), 32));
112 secure_wipe(block);
113 mac.update(aad);
114 pad16(mac, aad.size());
115 mac.update(ciphertext);
116 pad16(mac, ciphertext.size());
117 std::array<std::uint8_t, 16> lengths{};
118 for (std::size_t i = 0; i < 8; ++i) {
119 lengths[i] = static_cast<std::uint8_t>(static_cast<std::uint64_t>(aad.size()) >> (8 * i));
120 lengths[8 + i] = static_cast<std::uint8_t>(static_cast<std::uint64_t>(ciphertext.size()) >> (8 * i));
121 }
122 mac.update(lengths);
123 return mac.finish();
124}

126// The two tags equal, in constant time: every byte compared, the result revealed once
127constexpr bool tags_equal(const tag& a, std::span<const std::uint8_t, 16> b) noexcept
128{
129 std::uint64_t diff = 0;
130 for (std::size_t i = 0; i < 16; ++i)
131 diff |= static_cast<std::uint64_t>(a[i] ^ b[i]);
132 return ct::declassify(ct::mask_is_zero(diff)) != 0;
133}

135constexpr bool lengths_ok(std::size_t in, std::size_t out) noexcept { return in == out && in <= max_message; }

137// XChaCha20's subkey and 12-byte nonce from the key and the 24-byte nonce
138struct xparams {
139 secret_bytes<32> key;
140 std::array<std::uint8_t, 12> nonce{};
141};

143constexpr xparams xchacha(std::span<const std::uint8_t, 32> key, std::span<const std::uint8_t, 24> nonce) noexcept
144{
145 xparams x{secret_bytes<32>(crypto::detail::hchacha20(key, nonce.first<16>())), {}};
146 for (std::size_t i = 0; i < 8; ++i)
147 x.nonce[4 + i] = nonce[16 + i];
148 return x;
149}

161[[nodiscard]] constexpr result<tag, errc> chacha20_poly1305_seal(std::span<const std::uint8_t, 32> key,
162 std::span<const std::uint8_t, 12> nonce,
163 std::span<const std::uint8_t> aad,
164 std::span<const std::uint8_t> plaintext,
165 std::span<std::uint8_t> ciphertext) noexcept
166{
167 if (!detail::lengths_ok(plaintext.size(), ciphertext.size()))
168 return err(errc::invalid_length);
169 crypto::detail::chacha20_xor(key, 1, nonce, plaintext, ciphertext);
170 return ok(detail::compute_tag(key, nonce, aad, ciphertext));
171}

183[[nodiscard]] constexpr result<void, errc> chacha20_poly1305_open(std::span<const std::uint8_t, 32> key,
184 std::span<const std::uint8_t, 12> nonce,
185 std::span<const std::uint8_t> aad,
186 std::span<const std::uint8_t> ciphertext,
187 std::span<const std::uint8_t, 16> received,
188 std::span<std::uint8_t> plaintext) noexcept
189{
190 if (!detail::lengths_ok(ciphertext.size(), plaintext.size()))
191 return err(errc::invalid_length);
192 const tag expected = detail::compute_tag(key, nonce, aad, ciphertext);
193 if (!detail::tags_equal(expected, received)) {
194 for (auto& b : plaintext)
195 b = 0;
196 return err(errc::authentication_failed);
197 }
198 crypto::detail::chacha20_xor(key, 1, nonce, ciphertext, plaintext);
199 return ok();
200}
