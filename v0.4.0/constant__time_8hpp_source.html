113[[nodiscard]] constexpr std::uint64_t mask_from_bit(std::uint64_t bit) noexcept

121[[nodiscard]] constexpr std::uint64_t mask_is_zero(std::uint64_t x) noexcept
122{
123 // (x | -x) has its top bit set exactly when x is not zero
124 const std::uint64_t nz = (x | (std::uint64_t{0} - x)) >> 63;
125 return mask_from_bit(nz ^ 1);
126}

138[[nodiscard]] constexpr std::uint64_t mask_less(std::uint64_t a, std::uint64_t b) noexcept
139{
140 // The borrow out of a - b, computed without a comparison
141 const std::uint64_t borrow = ((~a & b) | ((~a | b) & (a - b))) >> 63;
142 return mask_from_bit(borrow);
143}

153[[nodiscard]] constexpr T select(std::uint64_t mask, T a, T b) noexcept
154{
155 const T m = static_cast<T>(value_barrier(mask));
156 return static_cast<T>(b ^ (m & (a ^ b)));

167{
168 const T m = static_cast<T>(value_barrier(mask));
169 const T t = static_cast<T>(m & (a ^ b));
170 a = static_cast<T>(a ^ t);

178[[nodiscard]] constexpr bool equal(std::span<const std::uint8_t> a, std::span<const std::uint8_t> b) noexcept
179{
180 if (a.size() != b.size())
181 return false;
182 std::uint64_t acc = 0;
183 for (std::size_t i = 0; i < a.size(); ++i)
184 acc |= static_cast<std::uint64_t>(a[i] ^ b[i]);
185 return mask_is_zero(acc) != 0;
186}

191[[nodiscard]] constexpr bool is_zero(std::span<const std::uint8_t> a) noexcept
192{
193 std::uint64_t acc = 0;
194 for (const std::uint8_t x : a)
195 acc |= x;
196 return mask_is_zero(acc) != 0;
197}

204constexpr void conditional_copy(std::uint64_t mask, std::span<std::uint8_t> dst, std::span<const std::uint8_t> src) noexcept
205{
206 const auto m = static_cast<std::uint8_t>(value_barrier(mask));
207 for (std::size_t i = 0; i < dst.size() && i < src.size(); ++i)
208 dst[i] = static_cast<std::uint8_t>(dst[i] ^ (m & (dst[i] ^ src[i])));
