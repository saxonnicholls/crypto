238[[nodiscard]] constexpr signature sign(const private_key& key, std::span<const std::uint8_t, 32> hash) noexcept
239{
240 const auto order = scalar::order_bytes();
241 scalar d = key.to_scalar();
242 scalar z = scalar::from_bytes_reduced(hash);
243 secp256k1::rfc6979 nonces(key.reveal(), hash, order, {});
244 for (;;) {
245 const secret_bytes<32> kb = nonces.next();
246 std::uint64_t below = 0;
247 scalar k = scalar::from_bytes_ct(kb.reveal(), below); // RFC 6979 gives 1 to n - 1: below is set
248 std::uint64_t finite = 0;
249 const affine_point big_r = point::mul_base(k).to_affine_ct(finite);
250 std::uint64_t r_below = 0;
251 const scalar r = scalar::from_bytes_ct(big_r.x.to_bytes(), r_below); // r = x mod n
252 scalar k_inverse = k.inverse();
253 const scalar s = k_inverse * (z + r * d);
254 secure_wipe(k);
255 secure_wipe(k_inverse);
256 // r or s of zero, with probability about 2^-256: the next RFC 6979 nonce. No `continue` (MSVC's C3615)
257 if (!ct::declassify(r.is_zero_mask() | s.is_zero_mask())) {
258 secure_wipe(d);
259 secure_wipe(z);
260 return {r.to_bytes(), s.to_bytes()};
261 }
262 }
263}

272[[nodiscard]] constexpr result<void, errc> verify(const public_key& key, std::span<const std::uint8_t, 32> hash,
273 const signature& sig) noexcept
274{
275 const auto r = scalar::from_bytes(sig.r);
276 const auto s = scalar::from_bytes(sig.s);
277 if (!r || !s || r->is_zero() || s->is_zero())
278 return err(errc::invalid_signature);
279 const scalar w = s->inverse();
280 const scalar z = scalar::from_bytes_reduced(hash);
281 const point big_r = point::mul_add_public(z * w, *r * w, point(key.point()));
282 const auto a = big_r.to_affine();
283 if (!a)
284 return err(errc::invalid_signature);
285 std::uint64_t below = 0;
286 if (!(scalar::from_bytes_ct(a->x.to_bytes(), below) == *r))
287 return err(errc::invalid_signature);
288 return ok();
289}
