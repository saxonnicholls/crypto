184 constexpr std::size_t to_text(std::span<char, max_text_length> out) const noexcept
185 {
186 for (char& c : out)
187 c = 0;
188 std::uint64_t pos = 0;
189 for (std::size_t w = 0; w < count_; ++w) {
190 std::uint64_t packed = 0;
191 for (std::uint64_t i = 0; i < word_list_size; ++i)
192 packed = ct::select(ct::mask_equal(i, words_[w]), english_packed[i], packed);
193 std::uint64_t len = 0;
194 for (std::size_t b = 0; b < max_word_length; ++b)
195 len += ~ct::mask_is_zero((packed >> (8 * b)) & 0xff) & 1;
196 const std::uint64_t space_wanted = ct::mask_from_bit(w + 1 < count_ ? 1 : 0);
197 for (std::uint64_t j = 0; j < max_text_length; ++j) {
198 // j through the barrier: otherwise GCC rewrites the loop around k, whose start is secret
199 const std::uint64_t k = ct::value_barrier(j) - pos;
200 const std::uint64_t in_word = ct::mask_less(k, len);
201 // the k-th byte of the word, chosen by masks over all eight (no shift by a secret amount)
202 std::uint64_t byte = 0;
203 for (std::uint64_t b = 0; b < max_word_length; ++b)
204 byte |= ct::mask_equal(k, b) & ((packed >> (8 * b)) & 0xff);
205 const auto letter = static_cast<unsigned char>(byte);
206 const std::uint64_t at_space = ct::mask_equal(k, len) & space_wanted;
207 auto c = static_cast<unsigned char>(out[j]);
208 c = ct::select<unsigned char>(in_word, letter, c);
209 c = ct::select<unsigned char>(at_space, static_cast<unsigned char>(' '), c);
210 out[j] = static_cast<char>(c);
211 }
212 pos += len + (space_wanted & 1);
213 secure_wipe(packed);
214 }
215 return static_cast<std::size_t>(pos);
216 }

222 [[nodiscard]] friend constexpr bool operator==(const mnemonic& a, const mnemonic& b) noexcept
223 {
224 std::uint64_t diff = a.count_ ^ b.count_;
225 for (std::size_t i = 0; i < max_words; ++i)
226 diff |= static_cast<std::uint64_t>(a.words_[i] ^ b.words_[i]);
227 return ct::mask_is_zero(diff) != 0;
228 }

238 // The mnemonic for `count` words of bits (entropy, then the checksum), big-endian
239 static constexpr mnemonic from_bits(const std::array<std::uint8_t, max_entropy_size + 1>& bits, std::size_t count) noexcept
240 {
241 mnemonic m;
242 m.count_ = count;
243 for (std::size_t w = 0; w < count; ++w) {
244 std::uint32_t v = 0;
245 for (std::size_t b = 0; b < 11; ++b) {
246 const std::size_t pos = 11 * w + b; // public positions; the bits themselves are secret
247 v = (v << 1) | ((bits[pos / 8] >> (7 - pos % 8)) & 1u);
248 }
249 m.words_[w] = static_cast<std::uint16_t>(v);
250 }
251 return m;
252 }

266// The checksum bits for `size` bytes of entropy, left-aligned in a byte: the first size / 4 bits of SHA-256
267constexpr std::uint8_t checksum(std::span<const std::uint8_t> entropy) noexcept
268{
269 const auto h = crypto::detail::sha256::hash(entropy);
270 const std::size_t cs_bits = entropy.size() / 4;
271 return static_cast<std::uint8_t>(h[0] & static_cast<std::uint8_t>(0xff << (8 - cs_bits)));
272}

279[[nodiscard]] constexpr result<mnemonic, errc> from_entropy(std::span<const std::uint8_t> entropy) noexcept
280{
281 if (entropy.size() < 16 || entropy.size() > max_entropy_size || entropy.size() % 4 != 0)
282 return err(errc::invalid_mnemonic_length);
283 std::array<std::uint8_t, max_entropy_size + 1> bits{};
284 for (std::size_t i = 0; i < entropy.size(); ++i)
285 bits[i] = entropy[i];
286 bits[entropy.size()] = detail::checksum(entropy);
287 mnemonic m = mnemonic::from_bits(bits, entropy.size() * 3 / 4);
288 secure_wipe(bits);
289 return ok(std::move(m));
290}

295[[nodiscard]] constexpr secret_bytes<max_entropy_size> to_entropy(const mnemonic& m) noexcept
296{
297 std::array<std::uint8_t, max_entropy_size + 1> bits{};
298 for (std::size_t w = 0; w < m.size(); ++w)
299 for (std::size_t b = 0; b < 11; ++b) {
300 const std::size_t pos = 11 * w + b;
301 const auto bit = static_cast<std::uint8_t>((m.index(w) >> (10 - b)) & 1u);
302 bits[pos / 8] = static_cast<std::uint8_t>(bits[pos / 8] | (bit << (7 - pos % 8)));
303 }
304 std::array<std::uint8_t, max_entropy_size> e{};
305 for (std::size_t i = 0; i < m.entropy_size(); ++i)
306 e[i] = bits[i];
307 secure_wipe(bits);
308 return secret_bytes<max_entropy_size>(std::move(e));
309}

314[[nodiscard]] constexpr result<mnemonic, errc> parse(std::string_view text) noexcept
315{
316 if (text.empty() || text.size() > max_text_length)
317 return err(errc::invalid_mnemonic_length);
318 // One slot per possible word; on every space every slot is written, the right one with the finished word
319 std::array<std::uint64_t, max_words + 1> slot{};
320 std::uint64_t bad = 0, count = 0, len = 0, acc = 0, too_long = 0;
321 for (const char ch : text) {
322 const auto c = static_cast<std::uint64_t>(static_cast<unsigned char>(ch));
323 const std::uint64_t space = ct::mask_equal(c, ' ');
324 const std::uint64_t letter = ~ct::mask_less(c, 'a') & ~ct::mask_less('z', c);
325 bad |= ~(space | letter);
326 bad |= space & ct::mask_is_zero(len); // a leading or a second space
327 const std::uint64_t finished = ct::select(too_long, std::uint64_t{0}, acc); // 0 matches no word
328 for (std::uint64_t k = 0; k <= max_words; ++k)
329 slot[k] = ct::select(space & ct::mask_equal(k, count), finished, slot[k]);
330 const std::uint64_t fits = letter & ct::mask_less(len, max_word_length);
331 too_long = ct::select(space, std::uint64_t{0}, too_long | (letter & ~fits));
332 acc = ct::select(space, std::uint64_t{0}, acc | (fits & (c << (8 * (len & 7)))));
333 len = ct::select(space, std::uint64_t{0}, len + 1);
334 count += space & 1;
335 }
336 bad |= ct::mask_is_zero(len); // a trailing space
337 const std::uint64_t finished = ct::select(too_long, std::uint64_t{0}, acc);
338 for (std::uint64_t k = 0; k <= max_words; ++k)
339 slot[k] = ct::select(ct::mask_equal(k, count), finished, slot[k]);
340 ++count;
341 secure_wipe(acc);
342
343 errc failure{};
344 count = ct::declassify(count); // the number of words is public (as the text's length is)
345 if (ct::declassify(bad))
346 failure = errc::non_canonical;
347 else if (!detail::valid_word_count(count))
348 failure = errc::invalid_mnemonic_length;
349 mnemonic m;
350 m.count_ = failure == errc{} ? count : 0;
351 std::uint64_t unknown = 0;
352 for (std::size_t w = 0; w < m.count_; ++w) {
353 const std::uint64_t i = find_word_ct(slot[w]);
354 unknown |= ct::mask_equal(i, word_list_size);
355 m.words_[w] = static_cast<std::uint16_t>(i & 0x7ff);
356 }
357 secure_wipe(slot);
358 if (failure == errc{} && ct::declassify(unknown))
359 failure = errc::unknown_word;
360 if (failure != errc{})
361 return err(failure);
362
363 // The checksum: recompute it from the entropy and compare it with the last bits, with masks
364 std::array<std::uint8_t, max_entropy_size + 1> bits{};
365 for (std::size_t w = 0; w < m.count_; ++w)
366 for (std::size_t b = 0; b < 11; ++b) {
367 const std::size_t pos = 11 * w + b;
368 const auto bit = static_cast<std::uint8_t>((m.words_[w] >> (10 - b)) & 1u);
369 bits[pos / 8] = static_cast<std::uint8_t>(bits[pos / 8] | (bit << (7 - pos % 8)));
370 }
371 const std::size_t ent = m.entropy_size();
372 const std::uint8_t expected = detail::checksum(std::span<const std::uint8_t>(bits).first(ent));
373 const std::uint64_t mismatch = ~ct::mask_equal(expected, bits[ent]);
374 secure_wipe(bits);
375 if (ct::declassify(mismatch))
376 return err(errc::invalid_checksum);
377 return ok(std::move(m));
378}

385[[nodiscard]] inline result<secret_bytes<64>, errc> to_seed(const mnemonic& m, std::string_view passphrase = {},
386 passphrase_form form = passphrase_form::ascii) noexcept
387{
388 std::uint64_t high = 0;
389 for (const char c : passphrase)
390 high |= static_cast<unsigned char>(c) & 0x80u;
391 if (form == passphrase_form::ascii && ct::declassify(high))
392 return err(errc::non_canonical);
393 std::array<char, max_text_length> text{};
394 // The text's length (the sum of the word lengths) decides how many SHA-512 blocks PBKDF2 hashes, so it reaches
395 // the timing: about four or five bits of the words, declassified here on purpose (see "Known issues").
396 const std::size_t n = ct::declassify(m.to_text(text));
397 std::array<std::uint8_t, 64> seed{};
398 crypto::detail::pbkdf2_hmac_sha512(crypto::detail::text_bytes({text.data(), n}), crypto::detail::text_bytes("mnemonic"),
399 crypto::detail::text_bytes(passphrase), 2048, seed);
400 secure_wipe(text);
401 return ok(secret_bytes<64>(std::move(seed)));
402}
