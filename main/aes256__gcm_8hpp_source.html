85constexpr void gcm_ctr(const crypto::detail::aes256& aes, std::array<std::uint8_t, 16> counter, std::span<const std::uint8_t> in,
86 std::span<std::uint8_t> out) noexcept
87{
88 for (std::size_t off = 0; off < in.size(); off += 16) {
89 // inc32: the last four bytes, big-endian
90 for (std::size_t i = 16; i-- > 12;)
91 if (++counter[i] != 0) // the counter is public
92 break;
93 auto ks = aes.encrypt(counter);
94 const std::size_t n = in.size() - off < 16 ? in.size() - off : 16;
95 for (std::size_t i = 0; i < n; ++i)
96 out[off + i] = static_cast<std::uint8_t>(in[off + i] ^ ks[i]);
97 secure_wipe(ks);
98 }
99}

101constexpr void ghash_update(std::array<std::uint8_t, 16>& y, const std::array<std::uint8_t, 16>& h, std::span<const std::uint8_t> data) noexcept
102{
103 for (std::size_t off = 0; off < data.size(); off += 16) {
104 const std::size_t n = data.size() - off < 16 ? data.size() - off : 16;
105 for (std::size_t i = 0; i < n; ++i)
106 y[i] ^= data[off + i];
107 y = crypto::detail::ghash_mul(y, h);
108 }
109}

111constexpr tag gcm_tag(const crypto::detail::aes256& aes, const std::array<std::uint8_t, 16>& j0, std::span<const std::uint8_t> aad,
112 std::span<const std::uint8_t> ciphertext) noexcept
113{
114 auto h = aes.encrypt({});
115 std::array<std::uint8_t, 16> y{};
116 ghash_update(y, h, aad);
117 ghash_update(y, h, ciphertext);
118 std::array<std::uint8_t, 16> lengths{};
119 const std::uint64_t abits = static_cast<std::uint64_t>(aad.size()) * 8, cbits = static_cast<std::uint64_t>(ciphertext.size()) * 8;
120 for (std::size_t i = 0; i < 8; ++i) {
121 lengths[i] = static_cast<std::uint8_t>(abits >> (56 - 8 * i));
122 lengths[8 + i] = static_cast<std::uint8_t>(cbits >> (56 - 8 * i));
123 }
124 ghash_update(y, h, lengths);
125 auto ek = aes.encrypt(j0);
126 tag t{};
127 for (std::size_t i = 0; i < 16; ++i)
128 t[i] = static_cast<std::uint8_t>(ek[i] ^ y[i]);
129 secure_wipe(h);
130 secure_wipe(ek);
131 return t;
132}

153[[nodiscard]] constexpr result<tag, errc> aes256_gcm_seal(std::span<const std::uint8_t, 32> key, std::span<const std::uint8_t, 12> nonce,
154 std::span<const std::uint8_t> aad, std::span<const std::uint8_t> plaintext,
155 std::span<std::uint8_t> ciphertext) noexcept
156{
157 if (plaintext.size() != ciphertext.size() || plaintext.size() > detail::gcm_max_message)
158 return err(errc::invalid_length);
159 const crypto::detail::aes256 aes(key);
160 const auto j0 = detail::gcm_j0(nonce);
161 detail::gcm_ctr(aes, j0, plaintext, ciphertext);
162 return ok(detail::gcm_tag(aes, j0, aad, ciphertext));
163}

175[[nodiscard]] constexpr result<void, errc> aes256_gcm_open(std::span<const std::uint8_t, 32> key, std::span<const std::uint8_t, 12> nonce,
176 std::span<const std::uint8_t> aad, std::span<const std::uint8_t> ciphertext,
177 std::span<const std::uint8_t, 16> received,
178 std::span<std::uint8_t> plaintext) noexcept
179{
180 if (plaintext.size() != ciphertext.size() || ciphertext.size() > detail::gcm_max_message)
181 return err(errc::invalid_length);
182 const crypto::detail::aes256 aes(key);
183 const auto j0 = detail::gcm_j0(nonce);
184 if (!detail::tags_equal(detail::gcm_tag(aes, j0, aad, ciphertext), received)) {
185 for (auto& b : plaintext)
186 b = 0;
187 return err(errc::authentication_failed);
188 }
189 detail::gcm_ctr(aes, j0, ciphertext, plaintext);
190 return ok();
191}
