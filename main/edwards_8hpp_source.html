207 [[nodiscard]] static constexpr std::optional<point> decode(std::span<const std::uint8_t, 32> b) noexcept
208 {
209 if (!field_element::is_canonical(b))
210 return std::nullopt;
211 const field_element y = field_element::from_bytes(b);
212 const field_element one = field_element::from_u64(1);
213 const field_element yy = y.square();
214 std::uint64_t ok = 0;
215 field_element x = field_element::sqrt_ratio(yy - one, d() * yy + one, ok);
216 if (!ok)
217 return std::nullopt;
218 const std::uint64_t sign = b[31] >> 7;
219 if (x.is_zero() && sign == 1)
220 return std::nullopt;
221 if (x.is_negative() != sign)
222 x = -x;
223 return from_affine(x, y);
224 }

257 [[nodiscard]] static constexpr point mul_public(std::span<const std::uint8_t, 32> k, const point& p) noexcept
258 {
259 const auto t = table(p);
260 point r;
261 for (std::size_t i = 64; i-- > 0;) {
262 r = r.dbl().dbl().dbl().dbl();
263 const std::uint64_t nib = (k[i / 2] >> (4 * (i % 2))) & 0xf;
264 if (nib != 0)
265 r = r + t[nib];
266 }
267 return r;
268 }

273 [[nodiscard]] static constexpr std::array<point, 16> table(const point& p) noexcept
274 {
275 std::array<point, 16> t{};
276 t[1] = p;
277 for (std::size_t j = 2; j < 16; ++j)
278 t[j] = (j % 2 == 0) ? t[j / 2].dbl() : t[j - 1] + p;
279 return t;
280 }

288 [[nodiscard]] static constexpr point mul_window(const scalar& k, const std::array<point, 16>& t) noexcept
289 {
290 point r;
291 for (std::size_t i = 64; i-- > 0;) {
292 r = r.dbl().dbl().dbl().dbl();
293 const std::uint64_t nib = k.nibble(i);
294 point e;
295 for (std::size_t j = 0; j < 16; ++j)
296 e = select(ct::mask_equal(j, nib), t[j], e);
297 r = r + e;
298 }
299 return r;
