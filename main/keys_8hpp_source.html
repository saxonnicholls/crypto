83 [[nodiscard]] static constexpr result<public_key, errc> parse(std::span<const std::uint8_t> bytes) noexcept
84 {
85 if (bytes.size() != 33 && bytes.size() != 65)
86 return err(errc::invalid_length);
87 const auto x = field_element::from_bytes(bytes.subspan<1, 32>());
88 if (!x)
89 return err(errc::invalid_public_key);
90 if (bytes.size() == 33) {
91 if (bytes[0] != 0x02 && bytes[0] != 0x03)
92 return err(errc::invalid_public_key);
93 const auto p = decompress(*x, bytes[0] & 1);
94 if (!p)
95 return err(errc::invalid_public_key);
96 return ok(public_key(*p));
97 }
98 if (bytes[0] != 0x04)
99 return err(errc::invalid_public_key);
100 const auto y = field_element::from_bytes(bytes.subspan<33, 32>());
101 if (!y)
102 return err(errc::invalid_public_key);
103 const affine_point p{*x, *y};
104 if (!p.is_on_curve())
105 return err(errc::invalid_public_key);
106 return ok(public_key(p));
107 }

111 [[nodiscard]] constexpr std::array<std::uint8_t, 33> compressed() const noexcept
112 {
113 std::array<std::uint8_t, 33> out{};
114 out[0] = static_cast<std::uint8_t>(0x02 | p_.y.is_odd());
115 const auto x = p_.x.to_bytes();
116 for (std::size_t i = 0; i < 32; ++i)
117 out[1 + i] = x[i];
118 return out;
119 }
120
123 [[nodiscard]] constexpr std::array<std::uint8_t, 65> uncompressed() const noexcept
124 {
125 std::array<std::uint8_t, 65> out{};
126 out[0] = 0x04;
127 const auto xy64 = xy();
128 for (std::size_t i = 0; i < 64; ++i)
129 out[1 + i] = xy64[i];
130 return out;
131 }
132
135 [[nodiscard]] constexpr std::array<std::uint8_t, 64> xy() const noexcept
136 {
137 std::array<std::uint8_t, 64> out{};
138 const auto x = p_.x.to_bytes(), y = p_.y.to_bytes();
139 for (std::size_t i = 0; i < 32; ++i) {
140 out[i] = x[i];
141 out[32 + i] = y[i];
142 }
143 return out;
144 }

162class private_key {
163public:
167 [[nodiscard]] static constexpr result<private_key, errc> from_bytes(std::span<const std::uint8_t, 32> bytes) noexcept
168 {
169 std::uint64_t below = 0;
170 scalar d = scalar::from_bytes_ct(bytes, below);
171 const std::uint64_t valid = below & ~d.is_zero_mask();
172 secure_wipe(d);
173 if (!ct::declassify(valid)) // reveals only that the key is invalid
174 return err(errc::invalid_private_key);
175 return ok(private_key(secret_bytes<32>(bytes)));
176 }
177
181 [[nodiscard]] static constexpr result<private_key, errc> from_secret(secret_bytes<32>&& bytes) noexcept
182 {
183 std::uint64_t below = 0;
184 scalar d = scalar::from_bytes_ct(bytes.reveal(), below);
185 const std::uint64_t valid = below & ~d.is_zero_mask();
186 secure_wipe(d);
187 if (!ct::declassify(valid))
188 return err(errc::invalid_private_key);
189 return ok(private_key(std::move(bytes)));
190 }

194 [[nodiscard]] constexpr secp256k1::public_key public_key() const noexcept
195 {
196 scalar d = to_scalar();
197 std::uint64_t finite = 0;
198 const affine_point p = point::mul_base(d).to_affine_ct(finite); // finite: d is in [1, n - 1]
199 secure_wipe(d);
200 return secp256k1::public_key(p);
201 }
