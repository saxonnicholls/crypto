77inline void secure_wipe(void* p, std::size_t n) noexcept
78{
79 volatile auto* q = static_cast<volatile std::uint8_t*>(p);
80 while (n--)
81 *q++ = 0;
82#if defined(__GNUC__) || defined(__clang__)
83 __asm__ __volatile__("" : : "r"(p) : "memory");
84#endif
85}
