107// A strict DER reader over one buffer: a tag, a minimal definite length, then that many bytes
108class der_reader {
109public:
110 constexpr explicit der_reader(std::span<const std::uint8_t> b) noexcept : b_(b) {}
111
112 [[nodiscard]] constexpr bool done() const noexcept { return i_ == b_.size(); }
113
114 // The contents of the next element if it has the tag; std::nullopt otherwise
115 constexpr std::optional<std::span<const std::uint8_t>> next(std::uint8_t tag) noexcept
116 {
117 if (b_.size() - i_ < 2 || b_[i_] != tag)
118 return std::nullopt;
119 std::size_t j = i_ + 1;
120 std::size_t len = b_[j++];
121 if (len & 0x80) {
122 const std::size_t n = len & 0x7f;
123 if (n == 0 || n > 2 || b_.size() - j < n || b_[j] == 0)
124 return std::nullopt; // indefinite, too long for a key, or not minimal
125 len = 0;
126 for (std::size_t k = 0; k < n; ++k)
127 len = len << 8 | b_[j++];
128 if (len < 0x80)
129 return std::nullopt; // the long form where the short one fits
130 }
131 if (b_.size() - j < len)
132 return std::nullopt;
133 i_ = j + len;
134 return b_.subspan(j, len);
135 }
136
137private:
138 std::span<const std::uint8_t> b_;
139 std::size_t i_ = 0;
140};

142// A DER INTEGER's contents as an unsigned big-endian magnitude: non-negative and minimal
143constexpr std::optional<std::span<const std::uint8_t>> der_unsigned(std::span<const std::uint8_t> v) noexcept
144{
145 if (v.empty() || (v[0] & 0x80))
146 return std::nullopt;
147 if (v.size() > 1 && v[0] == 0 && !(v[1] & 0x80))
148 return std::nullopt;
149 return v[0] == 0 && v.size() > 1 ? v.subspan(1) : v;
150}

182class public_key {
183public:
189 [[nodiscard]] static constexpr result<public_key, errc> from_components(std::span<const std::uint8_t> n,
190 std::span<const std::uint8_t> e) noexcept
191 {
192 if ((!n.empty() && n[0] == 0) || (!e.empty() && e[0] == 0))
193 return err(errc::non_canonical);
194 return make(n, e);
195 }
196
200 [[nodiscard]] static constexpr result<public_key, errc> from_pkcs1_der(std::span<const std::uint8_t> der) noexcept
201 {
202 detail::der_reader outer(der);
203 const auto seq = outer.next(0x30);
204 if (!seq || !outer.done())
205 return err(errc::invalid_public_key);
206 detail::der_reader in(*seq);
207 const auto n = in.next(0x02);
208 const auto e = n ? in.next(0x02) : std::nullopt;
209 if (!e || !in.done())
210 return err(errc::invalid_public_key);
211 const auto nu = detail::der_unsigned(*n);
212 const auto eu = detail::der_unsigned(*e);
213 if (!nu || !eu)
214 return err(errc::invalid_public_key);
215 return make(*nu, *eu);
216 }
217
222 [[nodiscard]] static constexpr result<public_key, errc> from_spki_der(std::span<const std::uint8_t> der) noexcept
223 {
224 detail::der_reader outer(der);
225 const auto seq = outer.next(0x30);
226 if (!seq || !outer.done())
227 return err(errc::invalid_public_key);
228 if (!seq->empty() && (*seq)[0] == 0x02)
229 return from_pkcs1_der(der); // a bare RSAPublicKey: its first element is the modulus
230 detail::der_reader in(*seq);
231 const auto alg = in.next(0x30);
232 const auto bits = alg ? in.next(0x03) : std::nullopt;
233 if (!bits || !in.done())
234 return err(errc::invalid_public_key);
235 detail::der_reader a(*alg);
236 const auto oid = a.next(0x06);
237 const auto null = oid ? a.next(0x05) : std::nullopt;
238 if (!null || !null->empty() || !a.done() || oid->size() != detail::rsa_encryption_oid.size())
239 return err(errc::invalid_public_key);
240 for (std::size_t i = 0; i < oid->size(); ++i)
241 if ((*oid)[i] != detail::rsa_encryption_oid[i])
242 return err(errc::invalid_public_key);
243 if (bits->empty() || (*bits)[0] != 0)
244 return err(errc::invalid_public_key); // a whole number of bytes: no unused bits
245 return from_pkcs1_der(bits->subspan(1));
246 }
247
250 [[nodiscard]] constexpr std::size_t modulus_bits() const noexcept { return bits_; }
251
254 [[nodiscard]] constexpr std::size_t modulus_bytes() const noexcept { return (bits_ + 7) / 8; }
255
258 [[nodiscard]] constexpr std::uint64_t exponent() const noexcept { return e_; }
259
265 [[nodiscard]] constexpr result<void, errc> open(std::span<const std::uint8_t> sig, std::span<std::uint8_t> out) const noexcept
266 {
267 const std::size_t k = modulus_bytes();
268 if (sig.size() != k || out.size() < k)
269 return err(errc::invalid_signature);
270 const auto s = crypto::detail::wide_from_be(sig);
271 if (!s || !crypto::detail::wide_less(*s, mont_.modulus(), crypto::detail::mont_max_limbs))
272 return err(errc::invalid_signature);
273 crypto::detail::wide_to_be(mont_.pow(*s, e_), out.first(k));
274 return ok();
275 }
276
277private:
278 constexpr public_key(const crypto::detail::montgomery& m, std::uint64_t e, std::size_t bits) noexcept
279 : mont_(m), e_(e), bits_(bits)
280 {
281 }
282
283 static constexpr result<public_key, errc> make(std::span<const std::uint8_t> n, std::span<const std::uint8_t> e) noexcept
284 {
285 if (n.size() > max_modulus_bytes || e.empty() || e.size() > 8)
286 return err(errc::invalid_public_key);
287 const auto nw = crypto::detail::wide_from_be(n);
288 if (!nw)
289 return err(errc::invalid_public_key);
290 const std::size_t bits = crypto::detail::wide_bits(*nw);
291 std::uint64_t ev = 0;
292 for (const auto b : e)
293 ev = ev << 8 | b;
294 if (bits < min_modulus_bits || bits > max_modulus_bits || ev < 3 || (ev & 1) == 0)
295 return err(errc::invalid_public_key);
296 const auto m = crypto::detail::montgomery::make(*nw); // refuses an even modulus
297 if (!m)
298 return err(errc::invalid_public_key);
299 return ok(public_key(*m, ev, bits));
300 }
301
302 crypto::detail::montgomery mont_;
303 std::uint64_t e_ = 0;
304 std::size_t bits_ = 0;
305};

309// MGF1 with H (RFC 8017 appendix B.2.1), XORed into out
310template <class H>
311constexpr void mgf1_xor(std::span<const std::uint8_t, H::digest_size> seed, std::span<std::uint8_t> out) noexcept
312{
313 std::size_t done = 0;
314 for (std::uint32_t counter = 0; done < out.size(); ++counter) {
315 H h;
316 h.update(seed);
317 const std::array<std::uint8_t, 4> c{static_cast<std::uint8_t>(counter >> 24), static_cast<std::uint8_t>(counter >> 16),
318 static_cast<std::uint8_t>(counter >> 8), static_cast<std::uint8_t>(counter)};
319 h.update(c);
320 const auto block = h.finish();
321 for (std::size_t i = 0; i < block.size() && done < out.size(); ++i)
322 out[done++] ^= block[i];
323 }
324}

326// RSASSA-PKCS1-v1_5 verification over a hash (RFC 8017 section 8.2.2): EM rebuilt and compared whole
327template <class Scheme>
328constexpr result<void, errc> verify_pkcs1v15(const public_key& key, std::span<const std::uint8_t, Scheme::hash::digest_size> hash,
329 std::span<const std::uint8_t> sig) noexcept
330{
331 std::array<std::uint8_t, max_modulus_bytes> em{};
332 if (const auto r = key.open(sig, em); !r)
333 return r;
334 // EM = 0x00 0x01 0xFF ... 0xFF 0x00 DigestInfo H (at least 8 bytes of 0xFF: k >= 256 > t + 11 for every hash here)
335 const std::size_t k = key.modulus_bytes();
336 const std::size_t t = Scheme::digest_info.size() + hash.size();
337 std::size_t i = 0;
338 bool same = em[i++] == 0x00 && em[i++] == 0x01;
339 for (; i < k - t - 1; ++i)
340 same = same && em[i] == 0xff;
341 same = same && em[i++] == 0x00;
342 for (const auto b : Scheme::digest_info)
343 same = same && em[i++] == b;
344 for (const auto b : hash)
345 same = same && em[i++] == b;
346 if (!same)
347 return err(errc::invalid_signature);
348 return ok();
349}

351// RSASSA-PSS verification over a hash with MGF1 on the same hash (RFC 8017 sections 8.1.2 and 9.1.2)
352template <class H>
353constexpr result<void, errc> verify_pss(const public_key& key, std::span<const std::uint8_t, H::digest_size> hash,
354 std::span<const std::uint8_t> sig, std::size_t salt_length) noexcept
355{
356 constexpr std::size_t h_len = H::digest_size;
357 const std::size_t em_bits = key.modulus_bits() - 1;
358 const std::size_t em_len = (em_bits + 7) / 8;
359 if (salt_length > em_len || em_len < h_len + salt_length + 2)
360 return err(errc::invalid_length);
361 std::array<std::uint8_t, max_modulus_bytes> m{};
362 if (const auto r = key.open(sig, m); !r)
363 return r;
364 // EM is the last em_len bytes of m; when em_bits is a multiple of 8 the extra leading byte must be zero
365 const std::size_t k = key.modulus_bytes();
366 if (k != em_len && m[0] != 0)
367 return err(errc::invalid_signature);
368 const std::span<std::uint8_t> em(m.data() + (k - em_len), em_len);
369 if (em[em_len - 1] != 0xbc)
370 return err(errc::invalid_signature);
371 const std::size_t db_len = em_len - h_len - 1;
372 const std::span<std::uint8_t> db = em.first(db_len);
373 const std::span<const std::uint8_t, h_len> h(em.data() + db_len, h_len);
374 const std::size_t zero_bits = 8 * em_len - em_bits;
375 const auto top_mask = static_cast<std::uint8_t>(0xff >> zero_bits);
376 if (db[0] & ~top_mask)
377 return err(errc::invalid_signature);
378 std::array<std::uint8_t, h_len> h_copy{};
379 for (std::size_t i = 0; i < h_len; ++i)
380 h_copy[i] = h[i];
381 mgf1_xor<H>(h_copy, db);
382 db[0] &= top_mask;
383 const std::size_t ps_len = db_len - salt_length - 1;
384 for (std::size_t i = 0; i < ps_len; ++i)
385 if (db[i] != 0)
386 return err(errc::invalid_signature);
387 if (db[ps_len] != 0x01)
388 return err(errc::invalid_signature);
389 // H' = Hash(0x00 x 8 || mHash || salt)
390 H hp;
391 const std::array<std::uint8_t, 8> zeros{};
392 hp.update(zeros);
393 hp.update(hash);
394 hp.update(db.subspan(ps_len + 1, salt_length));
395 const auto expected = hp.finish();
396 for (std::size_t i = 0; i < h_len; ++i)
397 if (expected[i] != h_copy[i])
398 return err(errc::invalid_signature);
399 return ok();
400}
