101// A strict DER reader over one buffer: a tag, a minimal definite length, then that many bytes
102class der_reader {
103public:
104 constexpr explicit der_reader(std::span<const std::uint8_t> b) noexcept : b_(b) {}
105
106 [[nodiscard]] constexpr bool done() const noexcept { return i_ == b_.size(); }
107
108 // The contents of the next element if it has the tag; std::nullopt otherwise
109 constexpr std::optional<std::span<const std::uint8_t>> next(std::uint8_t tag) noexcept
110 {
111 if (b_.size() - i_ < 2 || b_[i_] != tag)
112 return std::nullopt;
113 std::size_t j = i_ + 1;
114 std::size_t len = b_[j++];
115 if (len & 0x80) {
116 const std::size_t n = len & 0x7f;
117 if (n == 0 || n > 2 || b_.size() - j < n || b_[j] == 0)
118 return std::nullopt; // indefinite, too long for a key, or not minimal
119 len = 0;
120 for (std::size_t k = 0; k < n; ++k)
121 len = len << 8 | b_[j++];
122 if (len < 0x80)
123 return std::nullopt; // the long form where the short one fits
124 }
125 if (b_.size() - j < len)
126 return std::nullopt;
127 i_ = j + len;
128 return b_.subspan(j, len);
129 }
130
131private:
132 std::span<const std::uint8_t> b_;
133 std::size_t i_ = 0;
134};

136// A DER INTEGER's contents as an unsigned big-endian magnitude: non-negative and minimal
137constexpr std::optional<std::span<const std::uint8_t>> der_unsigned(std::span<const std::uint8_t> v) noexcept
138{
139 if (v.empty() || (v[0] & 0x80))
140 return std::nullopt;
141 if (v.size() > 1 && v[0] == 0 && !(v[1] & 0x80))
142 return std::nullopt;
143 return v[0] == 0 && v.size() > 1 ? v.subspan(1) : v;
144}

158class public_key {
159public:
165 [[nodiscard]] static constexpr result<public_key, errc> from_components(std::span<const std::uint8_t> n,
166 std::span<const std::uint8_t> e) noexcept
167 {
168 if ((!n.empty() && n[0] == 0) || (!e.empty() && e[0] == 0))
169 return err(errc::non_canonical);
170 return make(n, e);
171 }
172
176 [[nodiscard]] static constexpr result<public_key, errc> from_pkcs1_der(std::span<const std::uint8_t> der) noexcept
177 {
178 detail::der_reader outer(der);
179 const auto seq = outer.next(0x30);
180 if (!seq || !outer.done())
181 return err(errc::invalid_public_key);
182 detail::der_reader in(*seq);
183 const auto n = in.next(0x02);
184 const auto e = n ? in.next(0x02) : std::nullopt;
185 if (!e || !in.done())
186 return err(errc::invalid_public_key);
187 const auto nu = detail::der_unsigned(*n);
188 const auto eu = detail::der_unsigned(*e);
189 if (!nu || !eu)
190 return err(errc::invalid_public_key);
191 return make(*nu, *eu);
192 }
193
198 [[nodiscard]] static constexpr result<public_key, errc> from_spki_der(std::span<const std::uint8_t> der) noexcept
199 {
200 detail::der_reader outer(der);
201 const auto seq = outer.next(0x30);
202 if (!seq || !outer.done())
203 return err(errc::invalid_public_key);
204 if (!seq->empty() && (*seq)[0] == 0x02)
205 return from_pkcs1_der(der); // a bare RSAPublicKey: its first element is the modulus
206 detail::der_reader in(*seq);
207 const auto alg = in.next(0x30);
208 const auto bits = alg ? in.next(0x03) : std::nullopt;
209 if (!bits || !in.done())
210 return err(errc::invalid_public_key);
211 detail::der_reader a(*alg);
212 const auto oid = a.next(0x06);
213 const auto null = oid ? a.next(0x05) : std::nullopt;
214 if (!null || !null->empty() || !a.done() || oid->size() != detail::rsa_encryption_oid.size())
215 return err(errc::invalid_public_key);
216 for (std::size_t i = 0; i < oid->size(); ++i)
217 if ((*oid)[i] != detail::rsa_encryption_oid[i])
218 return err(errc::invalid_public_key);
219 if (bits->empty() || (*bits)[0] != 0)
220 return err(errc::invalid_public_key); // a whole number of bytes: no unused bits
221 return from_pkcs1_der(bits->subspan(1));
222 }
223
226 [[nodiscard]] constexpr std::size_t modulus_bits() const noexcept { return bits_; }
227
230 [[nodiscard]] constexpr std::size_t modulus_bytes() const noexcept { return (bits_ + 7) / 8; }
231
234 [[nodiscard]] constexpr std::uint64_t exponent() const noexcept { return e_; }
235
241 [[nodiscard]] constexpr result<void, errc> open(std::span<const std::uint8_t> sig, std::span<std::uint8_t> out) const noexcept
242 {
243 const std::size_t k = modulus_bytes();
244 if (sig.size() != k || out.size() < k)
245 return err(errc::invalid_signature);
246 const auto s = crypto::detail::wide_from_be(sig);
247 if (!s || !crypto::detail::wide_less(*s, mont_.modulus(), crypto::detail::mont_max_limbs))
248 return err(errc::invalid_signature);
249 crypto::detail::wide_to_be(mont_.pow(*s, e_), out.first(k));
250 return ok();
251 }
252
253private:
254 constexpr public_key(const crypto::detail::montgomery& m, std::uint64_t e, std::size_t bits) noexcept
255 : mont_(m), e_(e), bits_(bits)
256 {
257 }
258
259 static constexpr result<public_key, errc> make(std::span<const std::uint8_t> n, std::span<const std::uint8_t> e) noexcept
260 {
261 if (n.size() > max_modulus_bytes || e.empty() || e.size() > 8)
262 return err(errc::invalid_public_key);
263 const auto nw = crypto::detail::wide_from_be(n);
264 if (!nw)
265 return err(errc::invalid_public_key);
266 const std::size_t bits = crypto::detail::wide_bits(*nw);
267 std::uint64_t ev = 0;
268 for (const auto b : e)
269 ev = ev << 8 | b;
270 if (bits < min_modulus_bits || bits > max_modulus_bits || ev < 3 || (ev & 1) == 0)
271 return err(errc::invalid_public_key);
272 const auto m = crypto::detail::montgomery::make(*nw); // refuses an even modulus
273 if (!m)
274 return err(errc::invalid_public_key);
275 return ok(public_key(*m, ev, bits));
276 }
277
278 crypto::detail::montgomery mont_;
279 std::uint64_t e_ = 0;
280 std::size_t bits_ = 0;
281};

285// MGF1 with SHA-256 (RFC 8017 appendix B.2.1), XORed into out
286constexpr void mgf1_sha256_xor(std::span<const std::uint8_t, 32> seed, std::span<std::uint8_t> out) noexcept
287{
288 std::size_t done = 0;
289 for (std::uint32_t counter = 0; done < out.size(); ++counter) {
290 crypto::detail::sha256 h;
291 h.update(seed);
292 const std::array<std::uint8_t, 4> c{static_cast<std::uint8_t>(counter >> 24), static_cast<std::uint8_t>(counter >> 16),
293 static_cast<std::uint8_t>(counter >> 8), static_cast<std::uint8_t>(counter)};
294 h.update(c);
295 const auto block = h.finish();
296 for (std::size_t i = 0; i < block.size() && done < out.size(); ++i)
297 out[done++] ^= block[i];
298 }
299}

310[[nodiscard]] constexpr result<void, errc> verify_pkcs1v15_sha256_prehashed(const public_key& key,
311 std::span<const std::uint8_t, 32> hash,
312 std::span<const std::uint8_t> sig) noexcept
313{
314 std::array<std::uint8_t, max_modulus_bytes> em{};
315 if (const auto r = key.open(sig, em); !r)
316 return r;
317 // EM = 0x00 0x01 0xFF ... 0xFF 0x00 DigestInfo H, rebuilt and compared whole
318 const std::size_t k = key.modulus_bytes();
319 const std::size_t t = detail::sha256_digest_info.size() + hash.size();
320 std::size_t i = 0;
321 bool same = em[i++] == 0x00 && em[i++] == 0x01;
322 for (; i < k - t - 1; ++i)
323 same = same && em[i] == 0xff;
324 same = same && em[i++] == 0x00;
325 for (const auto b : detail::sha256_digest_info)
326 same = same && em[i++] == b;
327 for (const auto b : hash)
328 same = same && em[i++] == b;
329 if (!same)
330 return err(errc::invalid_signature);
331 return ok();
332}

356[[nodiscard]] constexpr result<void, errc> verify_pss_sha256_prehashed(const public_key& key, std::span<const std::uint8_t, 32> hash,
357 std::span<const std::uint8_t> sig,
358 std::size_t salt_length = 32) noexcept
359{
360 constexpr std::size_t h_len = 32;
361 const std::size_t em_bits = key.modulus_bits() - 1;
362 const std::size_t em_len = (em_bits + 7) / 8;
363 if (em_len < h_len + salt_length + 2)
364 return err(errc::invalid_length);
365 std::array<std::uint8_t, max_modulus_bytes> m{};
366 if (const auto r = key.open(sig, m); !r)
367 return r;
368 // EM is the last em_len bytes of m; when em_bits is a multiple of 8 the extra leading byte must be zero
369 const std::size_t k = key.modulus_bytes();
370 if (k != em_len && m[0] != 0)
371 return err(errc::invalid_signature);
372 const std::span<std::uint8_t> em(m.data() + (k - em_len), em_len);
373 if (em[em_len - 1] != 0xbc)
374 return err(errc::invalid_signature);
375 const std::size_t db_len = em_len - h_len - 1;
376 const std::span<std::uint8_t> db = em.first(db_len);
377 const std::span<const std::uint8_t, h_len> h(em.data() + db_len, h_len);
378 const std::size_t zero_bits = 8 * em_len - em_bits;
379 const auto top_mask = static_cast<std::uint8_t>(0xff >> zero_bits);
380 if (db[0] & ~top_mask)
381 return err(errc::invalid_signature);
382 std::array<std::uint8_t, h_len> h_copy{};
383 for (std::size_t i = 0; i < h_len; ++i)
384 h_copy[i] = h[i];
385 detail::mgf1_sha256_xor(h_copy, db);
386 db[0] &= top_mask;
387 const std::size_t ps_len = db_len - salt_length - 1;
388 for (std::size_t i = 0; i < ps_len; ++i)
389 if (db[i] != 0)
390 return err(errc::invalid_signature);
391 if (db[ps_len] != 0x01)
392 return err(errc::invalid_signature);
393 // H' = SHA-256(0x00 x 8 || mHash || salt)
394 crypto::detail::sha256 hp;
395 const std::array<std::uint8_t, 8> zeros{};
396 hp.update(zeros);
397 hp.update(hash);
398 hp.update(db.subspan(ps_len + 1, salt_length));
399 const auto expected = hp.finish();
400 for (std::size_t i = 0; i < h_len; ++i)
401 if (expected[i] != h_copy[i])
402 return err(errc::invalid_signature);
403 return ok();
404}
