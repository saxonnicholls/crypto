241[[nodiscard]] constexpr result<void, errc> verify(const public_key& key, std::span<const std::uint8_t, 48> hash,
242 const signature& sig) noexcept
243{
244 const auto r = scalar::from_bytes(sig.r);
245 const auto s = scalar::from_bytes(sig.s);
246 if (!r || !s || r->is_zero() || s->is_zero())
247 return err(errc::invalid_signature);
248 const scalar w = s->inverse();
249 const scalar z = scalar::from_bytes_reduced(hash);
250 const point big_r = point::mul_add_public(z * w, *r * w, point(key.point()));
251 const auto a = big_r.to_affine();
252 if (!a)
253 return err(errc::invalid_signature);
254 std::uint64_t below = 0;
255 if (!(scalar::from_bytes_ct(a->x.to_bytes(), below) == *r))
256 return err(errc::invalid_signature);
257 return ok();
258}
