89[[nodiscard]] constexpr std::array<std::uint8_t, 4> base58check_checksum(std::span<const std::uint8_t> payload) noexcept
90{
91 const auto once = crypto::detail::sha256::hash(payload);
92 const auto twice = crypto::detail::sha256::hash(once);
93 return {twice[0], twice[1], twice[2], twice[3]};
94}

107template <std::size_t MaxBytes = base58_default_max_bytes>
108[[nodiscard]] constexpr result<base58check_text<MaxBytes>, errc>
109base58check_encode(std::span<const std::uint8_t> payload, const base58_alphabet& alphabet = bitcoin_base58) noexcept
110{
111 if (payload.size() > MaxBytes)
112 return err(errc::invalid_length);
113 std::array<std::uint8_t, MaxBytes + 4> full{};
114 for (std::size_t i = 0; i < payload.size(); ++i)
115 full[i] = payload[i];
116 const auto check = base58check_checksum(payload);
117 for (std::size_t i = 0; i < 4; ++i)
118 full[payload.size() + i] = check[i];
119 std::array<char, base58_encoded_capacity(MaxBytes + 4)> buffer{};
120 const auto n = base58_encode_to<MaxBytes + 4>(std::span<const std::uint8_t>(full).first(payload.size() + 4), buffer, alphabet);
121 secure_wipe(full);
122 if (!n) {
123 secure_wipe(buffer);
124 return err(n.unwrap_err());
125 }
126 base58check_text<MaxBytes> text;
127 for (std::size_t i = 0; i < *n; ++i)
128 (void)text.try_push_back(buffer[i]);
129 secure_wipe(buffer);
130 return ok(text);
131}

150template <std::size_t MaxBytes = base58_default_max_bytes>
151[[nodiscard]] constexpr result<containers::inplace_vector<std::uint8_t, MaxBytes>, errc>
152base58check_decode(std::string_view text, const base58_alphabet& alphabet = bitcoin_base58) noexcept
153{
154 std::array<std::uint8_t, MaxBytes + 4> raw{};
155 const auto n = base58_decode_to<MaxBytes + 4>(text, raw, alphabet);
156 if (!n) {
157 secure_wipe(raw);
158 return err(n.unwrap_err());
159 }
160 if (*n < 4) {
161 secure_wipe(raw);
162 return err(errc::invalid_length);
163 }
164 const std::size_t body = *n - 4;
165 const auto check = base58check_checksum(std::span<const std::uint8_t>(raw).first(body));
166 if (!ct::equal(check, std::span<const std::uint8_t>(raw).subspan(body, 4))) {
167 secure_wipe(raw);
168 return err(errc::invalid_checksum);
169 }
170 containers::inplace_vector<std::uint8_t, MaxBytes> payload;
171 for (std::size_t i = 0; i < body; ++i)
172 (void)payload.try_push_back(raw[i]);
173 secure_wipe(raw);
174 return ok(payload);
175}

182template <std::size_t N>
183[[nodiscard]] constexpr result<std::array<std::uint8_t, N>, errc> base58check_decode_exact(std::string_view text,
184 const base58_alphabet& alphabet = bitcoin_base58) noexcept
185{
186 auto decoded = base58check_decode<N>(text, alphabet);
187 if (!decoded)
188 return err(decoded.unwrap_err());
189 if (decoded->size() != N) {
190 secure_wipe(*decoded);
191 return err(errc::invalid_length);
192 }
193 std::array<std::uint8_t, N> payload{};
194 for (std::size_t i = 0; i < N; ++i)
195 payload[i] = (*decoded)[i];
196 secure_wipe(*decoded);
197 return ok(payload);
198}
