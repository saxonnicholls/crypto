114[[nodiscard]] constexpr std::array<std::uint8_t, 32> scalar_mult(std::span<const std::uint8_t, 32> k,
115 std::span<const std::uint8_t, 32> u) noexcept
116{
117 using curve25519::field_element;
118 auto e = clamp(k);
119 const field_element x1 = field_element::from_bytes(u);
120 const field_element a24 = field_element::from_u64(121665);
121 field_element x2 = field_element::from_u64(1), z2{}, x3 = x1, z3 = field_element::from_u64(1);
122 std::uint64_t swap = 0;
123 for (int t = 254; t >= 0; --t) {
124 const std::uint64_t bit = (e[static_cast<std::size_t>(t) / 8] >> (t % 8)) & 1;
125 swap ^= bit;
126 const std::uint64_t mask = ct::mask_from_bit(swap);
127 field_element::conditional_swap(mask, x2, x3);
128 field_element::conditional_swap(mask, z2, z3);
129 swap = bit;
130 const field_element a = x2 + z2, aa = a.square(), b = x2 - z2, bb = b.square(), ee = aa - bb;
131 const field_element c = x3 + z3, d = x3 - z3, da = d * a, cb = c * b;
132 x3 = (da + cb).square();
133 z3 = x1 * (da - cb).square();
134 x2 = aa * bb;
135 z2 = ee * (aa + a24 * ee);
136 }
137 const std::uint64_t mask = ct::mask_from_bit(swap);
138 field_element::conditional_swap(mask, x2, x3);
139 field_element::conditional_swap(mask, z2, z3);
140 const auto out = (x2 * z2.invert()).to_bytes();
141 secure_wipe(e);
142 secure_wipe(x2);
143 secure_wipe(z2);
144 secure_wipe(x3);
145 secure_wipe(z3);
146 return out;
147}

162[[nodiscard]] constexpr snicholls::result<secret_bytes<32>, errc> shared_secret(const secret_bytes<32>& secret,
163 const public_key& peer) noexcept
164{
165 auto out = scalar_mult(secret.reveal(), peer.bytes);
166 if (ct::declassify(ct::is_zero(out)))
167 return snicholls::err(errc::small_order);
168 return snicholls::ok(secret_bytes<32>(std::move(out)));
169}
