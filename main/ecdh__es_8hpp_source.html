139template <std::size_t N>
140[[nodiscard]] constexpr result<secret_bytes<N>, errc> ecdh_es_x25519(const secret_bytes<32>& secret, const x25519::public_key& peer,
141 std::string_view algorithm, std::span<const std::uint8_t> apu = {},
142 std::span<const std::uint8_t> apv = {}) noexcept
143{
144 auto z = x25519::shared_secret(secret, peer);
