87struct affine_point {
88 field_element x;
89 field_element y;
90
93 [[nodiscard]] constexpr bool is_on_curve() const noexcept
94 {
95 const field_element three = field_element::from_u64(3);
96 return y.square() == x.square() * x - three * x + detail::b();
97 }
98
101 [[nodiscard]] constexpr std::array<std::uint8_t, 33> compressed() const noexcept
102 {
103 std::array<std::uint8_t, 33> out{};
104 out[0] = static_cast<std::uint8_t>(0x02 | y.is_odd());
105 const auto xb = x.to_bytes();
106 for (std::size_t i = 0; i < 32; ++i)
107 out[1 + i] = xb[i];
108 return out;
109 }
110
113 [[nodiscard]] constexpr std::array<std::uint8_t, 65> uncompressed() const noexcept
114 {
115 std::array<std::uint8_t, 65> out{};
116 out[0] = 0x04;
117 const auto xb = x.to_bytes(), yb = y.to_bytes();
118 for (std::size_t i = 0; i < 32; ++i) {
119 out[1 + i] = xb[i];
120 out[33 + i] = yb[i];
121 }
122 return out;
123 }
124
129 [[nodiscard]] static constexpr std::optional<affine_point> decode(std::span<const std::uint8_t> b) noexcept
130 {
131 if (b.size() == 65 && b[0] == 0x04) {
132 const auto x = field_element::from_bytes(b.subspan<1, 32>());
133 const auto y = field_element::from_bytes(b.subspan<33, 32>());
134 if (!x || !y)
135 return std::nullopt;
136 const affine_point p{*x, *y};
137 if (!p.is_on_curve())
138 return std::nullopt;
139 return p;
140 }
141 if (b.size() == 33 && (b[0] == 0x02 || b[0] == 0x03)) {
142 const auto x = field_element::from_bytes(b.subspan<1, 32>());
143 if (!x)
144 return std::nullopt;
145 const field_element rhs = x->square() * *x - field_element::from_u64(3) * *x + detail::b();
146 std::uint64_t square = 0;
147 field_element y = rhs.sqrt_ct(square);
148 if (!square)
149 return std::nullopt;
150 if (y.is_odd() != (b[0] & 1u))
151 y = -y;
152 return affine_point{*x, y};
153 }
154 return std::nullopt;
155 }

161 [[nodiscard]] friend constexpr bool operator==(const affine_point& a, const affine_point& c) noexcept = default;
162};

190 [[nodiscard]] friend constexpr point operator+(const point& p, const point& q) noexcept
191 {
192 const field_element b = detail::b();

194 field_element t3 = p.x_ + p.y_, t4 = q.x_ + q.y_;
195 t3 = t3 * t4;
196 t4 = t0 + t1;
197 t3 = t3 - t4;
198 t4 = p.y_ + p.z_;
199 field_element x3 = q.y_ + q.z_;
200 t4 = t4 * x3;
201 x3 = t1 + t2;
202 t4 = t4 - x3;
203 x3 = p.x_ + p.z_;
204 field_element y3 = q.x_ + q.z_;
205 x3 = x3 * y3;
206 y3 = t0 + t2;
207 y3 = x3 - y3;
208 field_element z3 = b * t2;
209 x3 = y3 - z3;
210 z3 = x3 + x3;
211 x3 = x3 + z3;
212 z3 = t1 - x3;
213 x3 = t1 + x3;
214 y3 = b * y3;
215 t1 = t2 + t2;
216 t2 = t1 + t2;
217 y3 = y3 - t2;
218 y3 = y3 - t0;
219 t1 = y3 + y3;
220 y3 = t1 + y3;
221 t1 = t0 + t0;
222 t0 = t1 + t0;
223 t0 = t0 - t2;
224 t1 = t4 * y3;
225 t2 = t0 * y3;
226 y3 = x3 * z3;
227 y3 = y3 + t2;
228 x3 = t3 * x3;
229 x3 = x3 - t1;

231 t1 = t3 * t0;
232 z3 = z3 + t1;

298 [[nodiscard]] static constexpr point mul(const scalar& k, const point& p) noexcept
299 {
300 const auto limbs = k.canonical();
301 std::array<point, 16> table{};
302 table[1] = p;
303 for (std::size_t j = 2; j < 16; ++j)
304 table[j] = table[j - 1] + p;
305 point r;
306 for (std::size_t i = 64; i-- > 0;) {
307 r = r.dbl().dbl().dbl().dbl();
308 const std::uint64_t nib = (limbs[i / 16] >> (4 * (i % 16))) & 0xf;
309 point t;
310 for (std::size_t j = 0; j < 16; ++j)
311 t = select(ct::mask_equal(j, nib), table[j], t);
312 r = r + t;
313 }
314 return r;
315 }

327 [[nodiscard]] static constexpr point mul_add_public(const scalar& a, const scalar& b, const point& q) noexcept
328 {
329 const auto la = a.canonical(), lb = b.canonical();
330 std::array<point, 16> tg{}, tq{};
331 tg[1] = base();
332 tq[1] = q;
333 for (std::size_t j = 2; j < 16; ++j) {
334 tg[j] = tg[j - 1] + tg[1];
335 tq[j] = tq[j - 1] + tq[1];
336 }
337 point r;
338 for (std::size_t i = 64; i-- > 0;) {
339 r = r.dbl().dbl().dbl().dbl();
340 if (const auto na = (la[i / 16] >> (4 * (i % 16))) & 0xf)
341 r = r + tg[na];
342 if (const auto nb = (lb[i / 16] >> (4 * (i % 16))) & 0xf)
343 r = r + tq[nb];
344 }
345 return r;
346 }
