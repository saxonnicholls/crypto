101constexpr std::uint32_t bech32_constant(bech32_variant v) noexcept { return v == bech32_variant::bech32 ? 1u : 0x2bc830a3u; }

103constexpr std::uint32_t polymod_step(std::uint32_t chk, std::uint8_t value) noexcept
104{
105 constexpr std::uint32_t generator[5] = {0x3b6a57b2u, 0x26508e6du, 0x1ea119fau, 0x3d4233ddu, 0x2a1462b3u};
106 const std::uint32_t top = chk >> 25;
107 chk = ((chk & 0x1ffffffu) << 5) ^ value;
108 for (int i = 0; i < 5; ++i)
109 chk ^= ((top >> i) & 1) ? generator[i] : 0;
110 return chk;
111}

113// The checksum state after the expanded HRP: high bits of each character, a zero, low bits of each character
114constexpr std::uint32_t polymod_hrp(std::string_view hrp) noexcept
115{
116 std::uint32_t chk = 1;
117 for (const char c : hrp)
118 chk = polymod_step(chk, static_cast<std::uint8_t>(static_cast<unsigned char>(c) >> 5));
119 chk = polymod_step(chk, 0);
120 for (const char c : hrp)
121 chk = polymod_step(chk, static_cast<std::uint8_t>(static_cast<unsigned char>(c) & 31));
122 return chk;
123}

171template <std::size_t MaxLength = bech32_max_length>
172[[nodiscard]] constexpr result<bech32_text<MaxLength>, errc> bech32_encode(std::string_view hrp, std::span<const std::uint8_t> values,
173 bech32_variant variant) noexcept
174{
175 if (hrp.empty() || hrp.size() > 83)
176 return err(errc::invalid_hrp);
177 if (hrp.size() + 1 + values.size() + 6 > MaxLength)
178 return err(errc::invalid_length);
179 std::array<char, 83> lower{};
180 for (std::size_t i = 0; i < hrp.size(); ++i) {
181 const auto c = static_cast<unsigned char>(hrp[i]);
182 if (c < 33 || c > 126)
183 return err(errc::invalid_hrp);
184 lower[i] = detail::to_lower(hrp[i]);
185 }
186 const std::string_view h(lower.data(), hrp.size());
187 std::uint32_t chk = detail::polymod_hrp(h);
188 bech32_text<MaxLength> text;
189 for (const char c : h)
190 (void)text.try_push_back(c);
191 (void)text.try_push_back('1');
192 for (const std::uint8_t v : values) {
193 if (v >= 32)
194 return err(errc::invalid_character);
195 chk = detail::polymod_step(chk, v);
196 (void)text.try_push_back(bech32_charset[v]);
197 }
198 for (int i = 0; i < 6; ++i)
199 chk = detail::polymod_step(chk, 0);
200 chk ^= detail::bech32_constant(variant);
201 for (int i = 0; i < 6; ++i)
202 (void)text.try_push_back(bech32_charset[(chk >> (5 * (5 - i))) & 31]);
203 return ok(text);
204}

210template <std::size_t MaxLength = bech32_max_length>
211[[nodiscard]] constexpr result<bech32_decoded<MaxLength>, errc> bech32_decode(std::string_view text) noexcept
212{
213 if (text.size() > MaxLength)
214 return err(errc::invalid_length);
215 bool lower = false, upper = false;
216 for (const char ch : text) {
217 const auto c = static_cast<unsigned char>(ch);
218 if (c < 33 || c > 126)
219 return err(errc::invalid_character);
220 lower = lower || (c >= 'a' && c <= 'z');
221 upper = upper || (c >= 'A' && c <= 'Z');
222 }
223 if (lower && upper)
224 return err(errc::non_canonical);
225 const std::size_t sep = text.rfind('1');
226 if (sep == std::string_view::npos || sep == 0 || sep > 83)
227 return err(errc::invalid_hrp);
228 if (text.size() - sep - 1 < 6)
229 return err(errc::invalid_length);
230 bech32_decoded<MaxLength> out;
231 for (std::size_t i = 0; i < sep; ++i)
232 (void)out.hrp.try_push_back(detail::to_lower(text[i]));
233 std::uint32_t chk = detail::polymod_hrp(std::string_view(out.hrp.data(), out.hrp.size()));
234 const std::size_t data_end = text.size() - 6;
235 for (std::size_t i = sep + 1; i < text.size(); ++i) {
236 const int v = detail::charset_value(detail::to_lower(text[i]));
237 if (v < 0)
238 return err(errc::invalid_character);
239 chk = detail::polymod_step(chk, static_cast<std::uint8_t>(v));
240 if (i < data_end)
241 (void)out.values.try_push_back(static_cast<std::uint8_t>(v));
242 }
243 if (chk == detail::bech32_constant(bech32_variant::bech32))
244 out.variant = bech32_variant::bech32;
245 else if (chk == detail::bech32_constant(bech32_variant::bech32m))
246 out.variant = bech32_variant::bech32m;
247 else
248 return err(errc::invalid_checksum);
249 return ok(out);
250}

260template <unsigned From, unsigned To, bool Pad, std::size_t N>
261 requires(From >= 1 && From <= 8 && To >= 1 && To <= 8)
262[[nodiscard]] constexpr result<containers::inplace_vector<std::uint8_t, N>, errc> convert_bits(std::span<const std::uint8_t> in) noexcept
263{
264 constexpr std::uint32_t max_value = (1u << To) - 1;
265 std::uint32_t acc = 0;
266 unsigned bits = 0;
267 containers::inplace_vector<std::uint8_t, N> out;
268 for (const std::uint8_t v : in) {
269 if constexpr (From < 8) { // a byte is always below 2^8 (and MSVC warns, C4333, on shifting it by 8)
270 if ((v >> From) != 0)
271 return err(errc::invalid_character);
272 }
273 acc = ((acc << From) | v) & 0xffffffu;
274 bits += From;
275 while (bits >= To) {
276 bits -= To;
277 if (!out.try_push_back(static_cast<std::uint8_t>((acc >> bits) & max_value)))
278 return err(errc::invalid_length);
279 }
280 }
281 if constexpr (Pad) {
282 if (bits > 0 && !out.try_push_back(static_cast<std::uint8_t>((acc << (To - bits)) & max_value)))
283 return err(errc::invalid_length);
284 } else {
285 if (bits >= From || ((acc << (To - bits)) & max_value) != 0)
286 return err(errc::non_canonical);
287 }
288 return ok(out);
289}

315[[nodiscard]] constexpr result<bech32_text<>, errc> segwit_encode(std::string_view hrp, std::uint8_t version,
316 std::span<const std::uint8_t> program) noexcept
317{
318 if (version > 16)
319 return err(errc::invalid_witness_version);
320 if (program.size() < 2 || program.size() > 40 || (version == 0 && program.size() != 20 && program.size() != 32))
321 return err(errc::invalid_witness_program);
322 auto five = convert_bits<8, 5, true, 64>(program);
323 if (!five)
324 return err(five.unwrap_err());
325 std::array<std::uint8_t, 65> values{};
326 values[0] = version;
327 for (std::size_t i = 0; i < five->size(); ++i)
328 values[i + 1] = (*five)[i];
329 return bech32_encode(hrp, std::span<const std::uint8_t>(values).first(five->size() + 1),
330 version == 0 ? bech32_variant::bech32 : bech32_variant::bech32m);
331}

337[[nodiscard]] constexpr result<witness_program, errc> segwit_decode(std::string_view hrp, std::string_view address) noexcept
338{
339 auto d = bech32_decode(address);
340 if (!d)
341 return err(d.unwrap_err());
342 if (d->hrp.size() != hrp.size())
343 return err(errc::invalid_hrp);
344 for (std::size_t i = 0; i < hrp.size(); ++i)
345 if (d->hrp[i] != detail::to_lower(hrp[i]))
346 return err(errc::invalid_hrp);
347 if (d->values.size() == 0)
348 return err(errc::invalid_witness_program);
349 const std::uint8_t version = d->values[0];
350 if (version > 16)
351 return err(errc::invalid_witness_version);
352 if ((version == 0) != (d->variant == bech32_variant::bech32))
353 return err(errc::invalid_checksum);
354 auto program = convert_bits<5, 8, false, 64>(std::span<const std::uint8_t>(d->values.data(), d->values.size()).subspan(1));
355 if (!program)
356 return err(program.unwrap_err());
357 if (program->size() < 2 || program->size() > 40 || (version == 0 && program->size() != 20 && program->size() != 32))
358 return err(errc::invalid_witness_program);
359 witness_program w;
360 w.version = version;
361 for (const std::uint8_t b : *program)
362 (void)w.program.try_push_back(b);
363 return ok(w);
364}
