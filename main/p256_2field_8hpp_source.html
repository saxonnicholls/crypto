170 [[nodiscard]] constexpr field_element sqrt_ct(std::uint64_t& is_square) const noexcept
171 {
172 crypto::detail::limbs4 e = P().m; // (p + 1) / 4: p + 1 = 2^256 - 2^224 + 2^192 + 2^96
173 crypto::detail::limbs4 one{1, 0, 0, 0};
174 crypto::detail::add4(e, e, one);
175 e = {(e[0] >> 2) | (e[1] << 62), (e[1] >> 2) | (e[2] << 62), (e[2] >> 2) | (e[3] << 62), e[3] >> 2};
176 const field_element r(crypto::detail::mont256_pow(v_, e, P()));
177 is_square = crypto::detail::mask_equal4(r.square().v_, v_);
