130 [[nodiscard]] static constexpr result<derivation_path, errc> parse(std::string_view text) noexcept
131 {
132 derivation_path p;
133 if (text.empty() || text[0] != 'm')
134 return err(errc::invalid_path);
135 std::size_t i = 1;
136 while (i < text.size()) {
137 if (text[i] != '/' || ++i == text.size())
138 return err(errc::invalid_path);
139 const std::size_t start = i;
140 std::uint64_t v = 0;
141 while (i < text.size() && text[i] >= '0' && text[i] <= '9' && i - start < 10)
142 v = v * 10 + static_cast<std::uint64_t>(text[i++] - '0');
143 const std::size_t digits = i - start;
144 if (digits == 0 || (digits > 1 && text[start] == '0') || v >= hardened_bit ||
145 (i < text.size() && text[i] >= '0' && text[i] <= '9'))
146 return err(errc::invalid_path);
147 auto index = static_cast<std::uint32_t>(v);
148 if (i < text.size() && (text[i] == '\'' || text[i] == 'h' || text[i] == 'H')) {
149 index = hardened(index);
150 ++i;
151 }
152 if (!p.push_back(index))
153 return err(errc::invalid_path);
154 }
155 return ok(p);
156 }
157
161 constexpr bool push_back(std::uint32_t index) noexcept
162 {
163 if (size_ == max_depth)
164 return false;
165 indices_[size_++] = index;
166 return true;
167 }
168
171 [[nodiscard]] constexpr std::size_t size() const noexcept { return size_; }
174 [[nodiscard]] constexpr bool empty() const noexcept { return size_ == 0; }
179 [[nodiscard]] constexpr std::uint32_t operator[](std::size_t i) const noexcept { return indices_[i]; }
182 [[nodiscard]] constexpr std::span<const std::uint32_t> indices() const noexcept { return {indices_.data(), size_}; }
185 [[nodiscard]] constexpr bool all_hardened() const noexcept
186 {
187 for (std::size_t i = 0; i < size_; ++i)
188 if (!is_hardened(indices_[i]))
189 return false;
190 return true;
191 }
192
196 constexpr std::size_t format(std::span<char, max_text_length> out) const noexcept
197 {
198 std::size_t n = 0;
199 out[n++] = 'm';
200 for (std::size_t i = 0; i < size_; ++i) {
201 out[n++] = '/';
202 std::uint32_t v = indices_[i] & ~hardened_bit;
203 char digits[10]{};
204 std::size_t d = 0;
205 do {
206 digits[d++] = static_cast<char>('0' + v % 10);
207 v /= 10;
208 } while (v != 0);
209 while (d > 0)
210 out[n++] = digits[--d];
211 if (is_hardened(indices_[i]))
212 out[n++] = '\'';
213 }
214 return n;
215 }
216
221 [[nodiscard]] friend constexpr bool operator==(const derivation_path& a, const derivation_path& b) noexcept
222 {
223 if (a.size_ != b.size_)
224 return false;
225 for (std::size_t i = 0; i < a.size_; ++i)
226 if (a.indices_[i] != b.indices_[i])
227 return false;
228 return true;
229 }
230
231private:
232 std::array<std::uint32_t, max_depth> indices_{};
233 std::size_t size_ = 0;
234};

243[[nodiscard]] constexpr result<derivation_path, errc> wallet_path(std::uint32_t purpose_number, std::uint32_t coin,
244 std::uint32_t account, std::uint32_t change,
245 std::uint32_t address_index) noexcept
246{
247 if (is_hardened(purpose_number) || is_hardened(coin) || is_hardened(account) || change > 1 || is_hardened(address_index))
248 return err(errc::invalid_path);
249 derivation_path p;
250 p.push_back(hardened(purpose_number));
251 p.push_back(hardened(coin));
252 p.push_back(hardened(account));
253 p.push_back(change);
254 p.push_back(address_index);
255 return ok(p);
256}
