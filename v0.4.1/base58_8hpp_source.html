98[[nodiscard]] constexpr base58_alphabet make_base58_alphabet(std::string_view symbols) noexcept
99{
100 base58_alphabet a;
101 for (auto& v : a.values)
102 v = -1;
103 if (symbols.size() != 58)
104 return a;
105 for (std::size_t i = 0; i < 58; ++i) {
106 const auto c = static_cast<unsigned char>(symbols[i]);
107 if (c < 33 || c > 126 || a.values[c] != -1)
108 return a;
109 a.symbols[i] = symbols[i];
110 a.values[c] = static_cast<std::int8_t>(i);
111 }
112 a.valid = true;
113 return a;
114}

138template <std::size_t MaxBytes = base58_default_max_bytes>
139[[nodiscard]] constexpr result<std::size_t, errc> base58_encode_to(std::span<const std::uint8_t> bytes, std::span<char> out,
140 const base58_alphabet& alphabet = bitcoin_base58) noexcept
141{
142 if (bytes.size() > MaxBytes)
143 return err(errc::invalid_length);
144 std::size_t zeros = 0;
145 while (zeros < bytes.size() && bytes[zeros] == 0) { // one leading symbol per leading zero byte
146 if (zeros < out.size())
147 out[zeros] = alphabet.symbols[0];
148 ++zeros;
149 }
150 if (zeros > out.size())
151 return err(errc::invalid_length);
152 // Base-58 digits of the rest, least significant first, built in out[zeros...]
153 const std::span<char> digits = out.subspan(zeros);
154 std::size_t length = 0;
155 for (std::size_t i = zeros; i < bytes.size(); ++i) {
156 unsigned carry = bytes[i];
157 for (std::size_t j = 0; j < length; ++j) {
158 carry += static_cast<unsigned>(static_cast<unsigned char>(digits[j])) << 8;
159 digits[j] = static_cast<char>(carry % 58);
160 carry /= 58;
161 }
162 while (carry != 0) {
163 if (length == digits.size())
164 return err(errc::invalid_length);
165 digits[length++] = static_cast<char>(carry % 58);
166 carry /= 58;
167 }
168 }
169 // Most significant first, each digit mapped to its symbol, in one two-ended pass (one loop, which GCC 13's
170 // vectoriser does not turn into stores it cannot bound)
171 std::size_t lo = 0, hi = length;
172 while (lo + 1 < hi) {
173 --hi;
174 const auto a = static_cast<unsigned char>(digits[lo]), b = static_cast<unsigned char>(digits[hi]);
175 digits[lo] = alphabet.symbols[b];
176 digits[hi] = alphabet.symbols[a];
177 ++lo;
178 }
179 if (lo + 1 == hi)
180 digits[lo] = alphabet.symbols[static_cast<unsigned char>(digits[lo])];
181 return ok(zeros + length);
182}

190template <std::size_t MaxBytes = base58_default_max_bytes>
191[[nodiscard]] constexpr result<std::size_t, errc> base58_decode_to(std::string_view text, std::span<std::uint8_t> out,
192 const base58_alphabet& alphabet = bitcoin_base58) noexcept
193{
194 if (text.size() > base58_encoded_capacity(MaxBytes))
195 return err(errc::invalid_length);
196 std::size_t zeros = 0;
197 while (zeros < text.size() && text[zeros] == alphabet.symbols[0])
198 ++zeros;
199 if (zeros > out.size())
200 return err(errc::invalid_length);
201 const std::span<std::uint8_t> bytes = out.subspan(zeros);
202 std::size_t length = 0; // bytes of the rest, least significant first
203 for (std::size_t i = zeros; i < text.size(); ++i) {
204 const auto c = static_cast<unsigned char>(text[i]);
205 const int digit = c < 128 ? alphabet.values[c] : -1;
206 if (digit < 0)
207 return err(errc::invalid_character);
208 unsigned carry = static_cast<unsigned>(digit);
209 for (std::size_t j = 0; j < length; ++j) {
210 carry += bytes[j] * 58u;
211 bytes[j] = static_cast<std::uint8_t>(carry & 0xff);
212 carry >>= 8;
213 }
214 while (carry != 0) {
215 if (length == bytes.size())
216 return err(errc::invalid_length);
217 bytes[length++] = static_cast<std::uint8_t>(carry & 0xff);
218 carry >>= 8;
219 }
220 }
221 // Most significant first. length never exceeds bytes.size() (the loop above refuses to grow past it), but GCC
222 // cannot carry that through the loop, and bounded by length alone this reversal is vectorised by GCC 13 on aarch64
223 // into stores past a one-byte buffer that it then reports (-Wstringop-overflow). Bounded by both, it is not.
224 const std::size_t n = length < bytes.size() ? length : bytes.size();
225 for (std::size_t i = 0; i < n / 2; ++i) {
226 const std::uint8_t t = bytes[i];
227 bytes[i] = bytes[n - 1 - i];
228 bytes[n - 1 - i] = t;
229 }
230 for (std::size_t i = 0; i < zeros; ++i)
231 out[i] = 0;
232 if (zeros + length > MaxBytes)
233 return err(errc::invalid_length);
234 return ok(zeros + length);
235}

248template <std::size_t MaxBytes = base58_default_max_bytes>
249[[nodiscard]] constexpr result<base58_text<MaxBytes>, errc> base58_encode(std::span<const std::uint8_t> bytes,
250 const base58_alphabet& alphabet = bitcoin_base58) noexcept
251{
252 std::array<char, base58_encoded_capacity(MaxBytes)> buffer{};
253 const auto n = base58_encode_to<MaxBytes>(bytes, buffer, alphabet);
254 if (!n)
255 return err(n.unwrap_err());
256 base58_text<MaxBytes> text;
257 for (std::size_t i = 0; i < *n; ++i)
258 (void)text.try_push_back(buffer[i]);
259 return ok(text);
260}
