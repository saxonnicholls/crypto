76 std::span<std::uint8_t> out) noexcept
77{
78 const std::size_t n = key.size() / 8;
79 if (key.size() % 8 != 0 || n < 2 || n > key_wrap_max_blocks || out.size() != key.size() + 8)

83 for (std::size_t i = 0; i < key.size(); ++i)
84 out[8 + i] = key[i];
85 for (std::size_t j = 0; j < 6; ++j)

112[[nodiscard]] constexpr result<void, errc> aes256_key_unwrap(std::span<const std::uint8_t, 32> kek, std::span<const std::uint8_t> wrapped,
113 std::span<std::uint8_t> out) noexcept
114{
115 if (wrapped.size() % 8 != 0 || wrapped.size() < 24 || wrapped.size() > 8 * (key_wrap_max_blocks + 1) ||

117 return err(errc::invalid_length);
118 const std::size_t n = wrapped.size() / 8 - 1;
119 const crypto::detail::aes256 aes(kek);
120 std::array<std::uint8_t, 8> a{};
121 for (std::size_t k = 0; k < 8; ++k)
122 a[k] = wrapped[k];
123 for (std::size_t i = 0; i < out.size(); ++i)
124 out[i] = wrapped[8 + i];
125 for (std::size_t j = 6; j-- > 0;)
126 for (std::size_t i = n; i >= 1; --i) {
127 const std::uint64_t t = static_cast<std::uint64_t>(n * j + i);
128 std::array<std::uint8_t, 16> b{};
129 for (std::size_t k = 0; k < 8; ++k) {
130 b[k] = static_cast<std::uint8_t>(a[k] ^ static_cast<std::uint8_t>(t >> (56 - 8 * k)));
131 b[8 + k] = out[8 * (i - 1) + k];
132 }
133 b = aes.decrypt(b);
134 for (std::size_t k = 0; k < 8; ++k) {
135 a[k] = b[k];

139 }

141 for (const auto v : a)
142 diff |= static_cast<std::uint64_t>(v ^ 0xa6);
143 if (ct::declassify(ct::mask_is_zero(diff)) == 0) {
144 for (auto& v : out)
145 v = 0;
