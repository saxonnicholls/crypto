113namespace detail {
114constexpr std::array<std::uint8_t, 4> be32(std::uint32_t v) noexcept
115{
116 return {static_cast<std::uint8_t>(v >> 24), static_cast<std::uint8_t>(v >> 16), static_cast<std::uint8_t>(v >> 8),
117 static_cast<std::uint8_t>(v)};
118}
119
120constexpr std::uint32_t read_be32(std::span<const std::uint8_t> b) noexcept
121{
122 return (std::uint32_t{b[0]} << 24) | (std::uint32_t{b[1]} << 16) | (std::uint32_t{b[2]} << 8) | std::uint32_t{b[3]};
123}

125constexpr std::array<std::uint8_t, 20> identifier_of(const secp256k1::public_key& k) noexcept
126{
127 return crypto::detail::hash160(k.compressed());
128}

130// version || depth || parent || child || chain || 33 key bytes
131constexpr std::array<std::uint8_t, 78> serialise(std::uint32_t version, std::uint8_t depth, const key_fingerprint& parent,
132 std::uint32_t child, std::span<const std::uint8_t, 32> chain,
133 std::span<const std::uint8_t, 33> key) noexcept
134{
135 std::array<std::uint8_t, 78> out{};
136 const auto v = be32(version), c = be32(child);
137 for (std::size_t i = 0; i < 4; ++i) {
138 out[i] = v[i];
139 out[5 + i] = parent[i];
140 out[9 + i] = c[i];
141 }
142 out[4] = depth;
143 for (std::size_t i = 0; i < 32; ++i)
144 out[13 + i] = chain[i];
145 for (std::size_t i = 0; i < 33; ++i)
146 out[45 + i] = key[i];
147 return out;
148}

150// The header fields every extended key shares, checked for consistency
151struct header {
152 std::uint32_t version;
153 std::uint8_t depth;
154 key_fingerprint parent;
155 std::uint32_t child;
156};
157
158constexpr result<header, errc> read_header(std::span<const std::uint8_t, 78> p) noexcept
159{
160 header h{read_be32(p.first<4>()), p[4], {p[5], p[6], p[7], p[8]}, read_be32(p.subspan<9, 4>())};
161 if (h.depth == 0 && (h.child != 0 || h.parent != key_fingerprint{}))
162 return err(errc::invalid_extended_key); // a master key has no parent and no index
163 return ok(h);
164}

169struct extended_public_key {
170 secp256k1::public_key key;
171 std::array<std::uint8_t, 32> chain;
172 std::uint8_t depth = 0;
173 key_fingerprint parent{};
174 std::uint32_t child_number = 0;

178 [[nodiscard]] constexpr std::array<std::uint8_t, 20> identifier() const noexcept { return detail::identifier_of(key); }
179
182 [[nodiscard]] constexpr key_fingerprint fingerprint() const noexcept
183 {
184 const auto id = identifier();
185 return {id[0], id[1], id[2], id[3]};
186 }
187
191 [[nodiscard]] constexpr result<extended_public_key, errc> derive_child(std::uint32_t index) const noexcept
192 {
193 if (is_hardened(index))
194 return err(errc::hardened_required);
195 crypto::detail::hmac_sha512 m(chain);
196 m.update(key.compressed());
197 m.update(detail::be32(index));
198 const auto i = m.finish();
199 const auto il = secp256k1::scalar::from_bytes(std::span<const std::uint8_t, 64>(i).first<32>());
200 if (!il)
201 return err(errc::invalid_derivation);
202 const auto child = (secp256k1::point::mul_base(*il) + secp256k1::point(key.point())).to_affine();
203 if (!child)
204 return err(errc::invalid_derivation);
205 extended_public_key out{secp256k1::public_key(*child), {}, static_cast<std::uint8_t>(depth + 1), fingerprint(), index};
206 for (std::size_t j = 0; j < 32; ++j)
207 out.chain[j] = i[32 + j];
208 return ok(out);
209 }
210
214 [[nodiscard]] constexpr result<extended_public_key, errc> derive(const derivation_path& path) const noexcept
215 {
216 extended_public_key k = *this;
217 for (const std::uint32_t index : path.indices()) {
218 auto next = k.derive_child(index);
219 if (!next)
220 return next;
221 k = *next;
222 }
223 return ok(k);
224 }

229 [[nodiscard]] constexpr std::array<std::uint8_t, 78> serialize(std::uint32_t version = bitcoin_mainnet.public_version) const noexcept
230 {
231 return detail::serialise(version, depth, parent, child_number, chain, key.compressed());
232 }

237 [[nodiscard]] constexpr extended_key_text to_text(const key_versions& versions = bitcoin_mainnet) const noexcept
238 {
239 return address::base58check_encode_exact(serialize(versions.public_version));
240 }

246 [[nodiscard]] static constexpr result<extended_public_key, errc> parse(std::span<const std::uint8_t, 78> payload,
247 const key_versions& versions = bitcoin_mainnet) noexcept
248 {
249 const auto h = detail::read_header(payload);
250 if (!h || h->version != versions.public_version)
251 return err(errc::invalid_extended_key);
252 const auto k = secp256k1::public_key::parse(payload.subspan<45, 33>());
253 if (!k || (payload[45] != 0x02 && payload[45] != 0x03))
254 return err(errc::invalid_extended_key);
255 extended_public_key out{*k, {}, h->depth, h->parent, h->child};
256 for (std::size_t j = 0; j < 32; ++j)
257 out.chain[j] = payload[13 + j];
258 return ok(out);
259 }

266 const key_versions& versions = bitcoin_mainnet) noexcept
267 {
268 const auto payload = address::base58check_decode_exact<78>(text);
269 if (!payload)
270 return err(payload.unwrap_err());
271 return parse(std::span<const std::uint8_t, 78>(*payload), versions);
272 }

288 [[nodiscard]] static constexpr result<extended_private_key, errc> from_seed(std::span<const std::uint8_t> seed) noexcept
289 {
290 if (seed.size() < 16 || seed.size() > 64)
291 return err(errc::invalid_seed);
292 auto i = crypto::detail::hmac_sha512::mac(crypto::detail::text_bytes("Bitcoin seed"), seed);
293 auto k = secp256k1::private_key::from_bytes(std::span<const std::uint8_t, 64>(i).first<32>());
294 if (!k) {
295 secure_wipe(i);
296 return err(errc::invalid_derivation);
297 }
298 extended_private_key out(std::move(*k), secret_bytes<32>(std::span<const std::uint8_t, 64>(i).last<32>()), 0, {}, 0);
299 secure_wipe(i);
300 return ok(std::move(out));
301 }

321 [[nodiscard]] constexpr result<extended_private_key, errc> derive_child(std::uint32_t index) const noexcept
322 {
323 const secp256k1::public_key parent_key = key_.public_key();
324 crypto::detail::hmac_sha512 m(chain_.reveal());
325 if (is_hardened(index)) {
326 const std::uint8_t zero = 0;
327 m.update(std::span<const std::uint8_t>(&zero, 1));
328 m.update(key_.reveal());
329 } else {
330 m.update(parent_key.compressed());
331 }
332 m.update(detail::be32(index));
333 auto i = m.finish();
334 std::uint64_t below = 0;
335 secp256k1::scalar il = secp256k1::scalar::from_bytes_ct(std::span<const std::uint8_t, 64>(i).first<32>(), below);
336 secp256k1::scalar kp = key_.to_scalar();
337 secp256k1::scalar child = il + kp;
338 const std::uint64_t valid = below & ~child.is_zero_mask();
339 auto child_bytes = child.to_bytes();
340 secure_wipe(il);
341 secure_wipe(kp);
342 secure_wipe(child);
343 if (!ct::declassify(valid)) { // probability below 2^-127, and public by BIP-32's rule
344 secure_wipe(i);
345 secure_wipe(child_bytes);
346 return err(errc::invalid_derivation);
347 }
348 auto k = secp256k1::private_key::from_secret(secret_bytes<32>(std::move(child_bytes)));
349 extended_private_key out(std::move(*k), secret_bytes<32>(std::span<const std::uint8_t, 64>(i).last<32>()),
350 static_cast<std::uint8_t>(depth_ + 1), fingerprint_of(parent_key), index);
351 secure_wipe(i);
352 return ok(std::move(out));
353 }
