97// BIP-340's tagged hash, SHA-256(SHA-256(tag) || SHA-256(tag) || ...): the state after the two tag blocks
98constexpr crypto::detail::sha256 tagged(std::string_view tag) noexcept
99{
100 crypto::detail::sha256 t;
101 t.update(tag);
102 const auto h = t.finish();
103 crypto::detail::sha256 out;
104 out.update(h);
105 out.update(h);
106 return out;
107}

109// d and P = d G with d negated (in constant time) when P's y is odd
110constexpr scalar even_y_key(const private_key& key, affine_point& p) noexcept
111{
112 scalar d = key.to_scalar();
113 std::uint64_t finite = 0;
114 p = point::mul_base(d).to_affine_ct(finite);
115 const std::uint64_t odd = ct::mask_from_bit(p.y.is_odd());
116 d = scalar::select(odd, -d, d);
117 p.y = field_element::select(odd, -p.y, p.y);
118 return d;
119}

140[[nodiscard]] constexpr schnorr_signature schnorr_sign(const private_key& key, std::span<const std::uint8_t> message,
141 std::span<const std::uint8_t, 32> aux) noexcept
142{
143 affine_point p;
144 scalar d = detail::even_y_key(key, p);
145 const auto px = p.x.to_bytes();
146 // t = bytes(d) xor hash_aux(a)
147 auto ha = detail::tagged("BIP0340/aux");
148 ha.update(aux);
149 auto t = ha.finish();
150 auto db = d.to_bytes();
151 for (std::size_t i = 0; i < 32; ++i)
152 t[i] ^= db[i];
153 for (;;) {
154 auto hn = detail::tagged("BIP0340/nonce");
155 hn.update(t);
156 hn.update(px);
157 hn.update(message);
158 auto rand = hn.finish();
159 scalar k = scalar::from_bytes_reduced(rand);
160 secure_wipe(rand);
161 // k of zero, with probability about 2^-256: BIP-340 fails here, this library tries once more with aux'. The
162 // loop has no `continue`: MSVC takes one in an endless loop for control reaching the end of a constexpr
163 // function (C3615).
164 if (!ct::declassify(k.is_zero_mask())) {
165 std::uint64_t finite = 0;
166 const affine_point big_r = point::mul_base(k).to_affine_ct(finite);
167 k = scalar::select(ct::mask_from_bit(big_r.y.is_odd()), -k, k);
168 const auto rx = big_r.x.to_bytes();
169 auto he = detail::tagged("BIP0340/challenge");
170 he.update(rx);
171 he.update(px);
172 he.update(message);
173 const scalar e = scalar::from_bytes_reduced(he.finish());
174 const auto s = (k + e * d).to_bytes();
175 schnorr_signature sig;
176 for (std::size_t i = 0; i < 32; ++i) {
177 sig.bytes[i] = rx[i];
178 sig.bytes[32 + i] = s[i];
179 }
180 secure_wipe(k);
181 secure_wipe(d);
182 secure_wipe(t);
183 secure_wipe(db);
184 return sig;
185 }
186 t[0] ^= 1;
187 }
188}

196[[nodiscard]] constexpr result<void, errc> schnorr_verify(const xonly_public_key& key, std::span<const std::uint8_t> message,
197 const schnorr_signature& sig) noexcept
198{
199 const auto px = field_element::from_bytes(key.bytes);
200 if (!px)
201 return err(errc::invalid_public_key);
202 const auto p = decompress(*px, 0);
203 if (!p)
204 return err(errc::invalid_public_key);
205 const std::span<const std::uint8_t, 64> b(sig.bytes);
206 const auto r = field_element::from_bytes(b.first<32>());
207 const auto s = scalar::from_bytes(b.last<32>());
208 if (!r || !s)
209 return err(errc::invalid_signature);
210 auto he = detail::tagged("BIP0340/challenge");
211 he.update(b.first<32>());
212 he.update(key.bytes);
213 he.update(message);
214 const scalar e = scalar::from_bytes_reduced(he.finish());
215 const auto big_r = point::mul_add_public(*s, -e, point(*p)).to_affine();
216 if (!big_r || big_r->y.is_odd() || !(big_r->x == *r))
217 return err(errc::invalid_signature);
218 return ok();
219}
