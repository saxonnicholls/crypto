256 [[nodiscard]] static constexpr point mul(const scalar& k, const point& p) noexcept
257 {
258 std::array<point, 16> table{};
259 table[1] = p;
260 for (std::size_t j = 2; j < 16; ++j)
261 table[j] = (j % 2 == 0) ? table[j / 2].dbl() : table[j - 1] + p;
262 point r;
263 for (std::size_t i = 64; i-- > 0;) {
264 r = r.dbl().dbl().dbl().dbl();
265 const std::uint64_t nib = k.nibble(i);
266 point t;
267 for (std::size_t j = 0; j < 16; ++j)
268 t = select(ct::mask_equal(j, nib), table[j], t);
269 r = r + t;
270 }
271 return r;
272 }

284 [[nodiscard]] static constexpr point mul_add_public(const scalar& a, const scalar& b, const point& q) noexcept
285 {
286 std::array<point, 16> tg{}, tq{};
287 tg[1] = base();
288 tq[1] = q;
289 for (std::size_t j = 2; j < 16; ++j) {
290 tg[j] = tg[j - 1] + tg[1];
291 tq[j] = tq[j - 1] + tq[1];
292 }
293 point r;
294 for (std::size_t i = 64; i-- > 0;) {
295 r = r.dbl().dbl().dbl().dbl();
296 if (const auto na = a.nibble(i))
297 r = r + tg[na];
298 if (const auto nb = b.nibble(i))
299 r = r + tq[nb];
300 }
301 return r;
